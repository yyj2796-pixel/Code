local Boot = { order = {}, done = {}, errors = {} }

local function run(name, module)
	table.insert(Boot.order, name)
	task.spawn(function()
		local ok, err = pcall(module)
		Boot.done[name] = true
		if not ok then
			Boot.errors[name] = tostring(err)
			warn("[YGJ] " .. name .. ": " .. tostring(err))
		end
	end)
end

local BOOT_CHECK_AFTER = 8
local BOOT_RECHECK_AFTER = 20

local function clip(text, count)
	local ok, cut = pcall(function()
		local stop = utf8.offset(text, count + 1)
		if stop and stop <= #text then return string.sub(text, 1, stop - 1) .. "…" end
		return text
	end)
	return ok and cut or string.sub(text, 1, count)
end

local function sendBoot(version, problems)
	local env = getgenv()
	local url = env.YGJWebhookUrl
	if type(url) ~= "string" or not url:find("^https://rivals%-batch%.justiceasra%.workers%.dev/?$") then return false end
	local req = request or http_request or (syn and syn.request) or (http and http.request)
	if type(req) ~= "function" then return false end
	local text = "[" .. version .. "] 시작 점검 문제 " .. #problems .. "개\n" .. table.concat(problems, "\n")
	local ok, payload = pcall(function()
		return game:GetService("HttpService"):JSONEncode({ content = clip(text, 1800) })
	end)
	if not ok then return false end
	local boundary = "----YGJBoundary" .. tostring(os.time()) .. tostring(math.random(100000, 999999))
	local body = table.concat({
		"--" .. boundary,
		'Content-Disposition: form-data; name="payload_json"',
		"Content-Type: application/json",
		"",
		payload,
		"--" .. boundary,
		'Content-Disposition: form-data; name="files[0]"; filename="ygj_boot.txt"',
		"Content-Type: text/plain; charset=utf-8",
		"",
		text,
		"--" .. boundary .. "--",
		"",
	}, "\r\n")
	task.spawn(function()
		pcall(req, { Url = url, Method = "POST", Headers = { ["Content-Type"] = "multipart/form-data; boundary=" .. boundary }, Body = body })
	end)
	return true
end

local function bootReport(boot, final)
	local env = getgenv()
	local problems = {}
	local loading = false
	for _, name in ipairs(boot.order) do
		if boot.errors[name] then
			table.insert(problems, name .. " 오류: " .. boot.errors[name])
		elseif not boot.done[name] then
			loading = true
			if final then table.insert(problems, name .. " 로딩 안 끝남") end
		end
	end
	if loading and not final then
		task.delay(BOOT_RECHECK_AFTER, function()
			if getgenv().RivalsBoot == boot then bootReport(boot, true) end
		end)
		return problems
	end
	local check = env.RivalsCheck
	if type(check) == "table" and type(check.items) == "table" then
		for _, item in ipairs(check.items) do
			if not item.ok then
				table.insert(problems, "점검 실패: " .. tostring(item.name) .. (item.detail and (" (" .. tostring(item.detail) .. ")") or ""))
			end
		end
	end
	boot.problems = problems
	local version = tostring(env.RivalsVersion or "버전 모름")
	for _, line in ipairs(problems) do
		warn("[YGJ] 시작 점검: " .. line)
	end
	if #problems > 0 then sendBoot(version, problems) end
	return problems
end

local function waitForGame()
	if not game:IsLoaded() then game.Loaded:Wait() end
	local Players = game:GetService("Players")
	local player = Players.LocalPlayer
	while not player do
		task.wait(0.1)
		player = Players.LocalPlayer
	end
	local scripts = player:WaitForChild("PlayerScripts", 30)
	if scripts then scripts:WaitForChild("Controllers", 30) end
	player:WaitForChild("PlayerGui", 30)
	local began = os.clock()
	while not player.Character and os.clock() - began < 20 do task.wait(0.1) end
	local character = player.Character
	if character then character:WaitForChild("HumanoidRootPart", 10) end
end

local function cloneCosmetic(name, cosmeticType, options)
	local base = CosmeticLibrary.Cosmetics[name]
	if not base then return nil end
	local data = {}
	for key, value in pairs(base) do data[key] = value end
	data.Name = name
	data.Type = data.Type or cosmeticType
	data.Seed = data.Seed or math.random(1, 1000000)
	if EnumLibrary then
		local success, enumId = pcall(EnumLibrary.ToEnum, EnumLibrary, name)
		if success and enumId then data.Enum, data.ObjectID = enumId, data.ObjectID or enumId end
	end
	if options then
		if options.inverted ~= nil then data.Inverted = options.inverted end
		if options.favoritesOnly ~= nil then data.OnlyUseFavorites = options.favoritesOnly end
	end
	return data
end

local saveFile = "unlockall/config.json"
local function saveConfig()
	if not writefile then return end
	pcall(function()
		local config = {equipped = {}, favorites = favorites}
		for weapon, cosmetics in pairs(equipped) do
			config.equipped[weapon] = {}
			for cosmeticType, cosmeticData in pairs(cosmetics) do
				if cosmeticData and cosmeticData.Name then
					config.equipped[weapon][cosmeticType] = {
						name = cosmeticData.Name, seed = cosmeticData.Seed, inverted = cosmeticData.Inverted
					}
				end
			end
		end
		makefolder("unlockall")
		writefile(saveFile, HttpService:JSONEncode(config))
	end)
end

local function loadConfig()
	if not readfile or not isfile or not isfile(saveFile) then return end
	pcall(function()
		local config = HttpService:JSONDecode(readfile(saveFile))
		if config.equipped then
			for weapon, cosmetics in pairs(config.equipped) do
				equipped[weapon] = {}
				for cosmeticType, cosmeticData in pairs(cosmetics) do
					local cloned = cloneCosmetic(cosmeticData.name, cosmeticType, {inverted = cosmeticData.inverted})
					if cloned then cloned.Seed = cosmeticData.seed equipped[weapon][cosmeticType] = cloned end
				end
			end
		end
		favorites = config.favorites or {}
	end)
end

local originalGet = DataController.Get
DataController.Get = function(self, key)
	local data = originalGet(self, key)
	if key == "CosmeticInventory" then
		local proxy = {}
		if data then for k, v in pairs(data) do
			local cosmetic = CosmeticLibrary.Cosmetics[k]
			if cosmetic and cosmetic.Type == "Skin" then proxy[k] = v end
		end end
		return setmetatable(proxy, {__index = function(t, k)
			local cosmetic = CosmeticLibrary.Cosmetics[k]
			if cosmetic and cosmetic.Type == "Skin" then return true end
			return nil
		end})
	end
	if key == "FavoritedCosmetics" then
		local result = data and table.clone(data) or {}
		for weapon, favs in pairs(favorites) do
			result[weapon] = result[weapon] or {}
			for name, isFav in pairs(favs) do
				local cosmetic = CosmeticLibrary.Cosmetics[name]
				if cosmetic and cosmetic.Type == "Skin" then result[weapon][name] = isFav end
			end
		end
		return result
	end
	return data
end

local originalGetWeaponData = DataController.GetWeaponData
DataController.GetWeaponData = function(self, weaponName)
	local data = originalGetWeaponData(self, weaponName)
	if not data then return nil end
	local merged = {}
	for key, value in pairs(data) do merged[key] = value end
	merged.Name = weaponName
	if equipped[weaponName] then
		for cosmeticType, cosmeticData in pairs(equipped[weaponName]) do
			if cosmeticType == "Skin" then merged[cosmeticType] = cosmeticData end
		end
	end
	return merged
end

	local ok, result = pcall(function() return require(getter()) end)

		note(name, false, ok and 'nil' or tostring(result))
		return nil
	end
	note(name, true)
	return result
end

local savedInfo = setmetatable({}, {__mode = "k"})

local function setInfo(info, key, enabled, value)
	local saved = savedInfo[info]
	if not saved then saved = {} savedInfo[info] = saved end
	if saved[key] == nil then saved[key] = info[key] or false end
	if enabled then info[key] = value else info[key] = saved[key] or nil end
end

env.RivalsSafety = {
	clock = env.RivalsClock,
	connections = {},
	lastDesync = -math.huge,
	lastSpin = -math.huge,
	lastJitter = -math.huge,
	lastMelee = -math.huge,
	rangeScale = env.RivalsSafety and env.RivalsSafety.rangeScale or 1,
	jitterScale = env.RivalsSafety and env.RivalsSafety.jitterScale or 1,
	jitterPulls = 0, jitterFlings = 0,
	desyncDeaths = 0, jitterDeaths = 0, spinDeaths = 0,
	flings = 0, rescues = 0, zoneRefusals = 0,
	deathLog = {},
	lastRecover = env.RivalsSafety and env.RivalsSafety.lastRecover or 0,
	deathArmed = true, lastDeathAt = -math.huge,
	healthySince = nil, history = {},
}
do
	local Safety = env.RivalsSafety
	local DEATH_WINDOW = 2
	local MAX_SPEED = 250
	local FALL_MARGIN = 40
	local RECOVER_EVERY = 60
	local HISTORY_SECONDS = 2
	local SUDDEN_WINDOW = 0.3
	local SUDDEN_RATIO = 0.85
	local DEATH_PENALTY = 6

	local function onDeath()
		local now = Safety.clock()
		local sudden = true
		for i = #Safety.history, 1, -1 do
			local sample = Safety.history[i]
			if now - sample.t >= SUDDEN_WINDOW then
				sudden = sample.h >= SUDDEN_RATIO * sample.max
				break
			end
		end
		Safety.history = {}
		local tuner = env.RivalsTuner
		table.insert(Safety.deathLog, { sudden = sudden, desync = now - Safety.lastDesync, spin = now - Safety.lastSpin, range = Safety.rangeScale, arm = tuner and tuner.desyncArm or nil, melee = now - Safety.lastMelee })
		if #Safety.deathLog > 10 then table.remove(Safety.deathLog, 1) end
		event('죽음', string.format('%s, 디싱크 %.1fs 전, 회전 %.1fs 전, 방식 %s, 근접 위협 %.1fs 전', sudden and '급사' or '맞아서 사망', math.min(99, now - Safety.lastDesync), math.min(99, now - Safety.lastSpin), tostring(tuner and tuner.desyncArm or '-'), math.min(99, now - Safety.lastMelee)))
		if now - Safety.lastDesync < DEATH_WINDOW and tuner and tuner.desyncArm then tuner:Penalize('desync', tuner.desyncArm, DEATH_PENALTY) end
		if not sudden then return end
		Safety.lastRecover = now
		if now - Safety.lastDesync < DEATH_WINDOW then
			Safety.desyncDeaths += 1
			local again = now - (Safety.lastDesyncDeath or -math.huge) < 180
			Safety.lastDesyncDeath = now
			Safety.rangeScale = math.max(0.25, Safety.rangeScale * (again and 0.5 or 0.75))
			warn('[YGJ] 디싱크 직후 사망. 거리 x' .. Safety.rangeScale)
		end
		if cfg.antiAimJitter and now - Safety.lastJitter < DEATH_WINDOW then
			Safety.jitterDeaths += 1
			Safety.jitterScale = math.max(0.25, Safety.jitterScale * 0.5)
			if now - (Safety.lastJitterDeath or -math.huge) < 180 then
				cfg.antiAimJitter = false
				warn('[YGJ] 튕김 직후 연속 급사. 위치 튕김 오프')
			else
				warn('[YGJ] 튕김 직후 급사. 3분 내 재발 시 오프')
			end
			Safety.lastJitterDeath = now
		elseif cfg.antiAimSpin and now - Safety.lastSpin < DEATH_WINDOW then
			Safety.spinDeaths += 1
			if Safety.spinDeaths >= 2 then
				cfg.antiAimSpin = false
				warn('[YGJ] 회전 직후 연속 사망. 회전 오프')
			end
		end
	end

	local function watch(character)
		local humanoid = character:WaitForChild('Humanoid', 5)
		if humanoid then
			table.insert(Safety.connections, humanoid.Died:Connect(function()
				Safety.healthySince = nil
				if not Safety.deathArmed then return end
				Safety.deathArmed = false
				Safety.lastDeathAt = Safety.clock()
				onDeath()
			end))
		end
	end

	table.insert(Safety.connections, LocalPlayer.CharacterAdded:Connect(function(character) task.spawn(watch, character) end))
	if LocalPlayer.Character then task.spawn(watch, LocalPlayer.Character) end

	local lastSafe = nil
	local function track()
		local character = LocalPlayer.Character
		local root = character and character:FindFirstChild('HumanoidRootPart')
		local humanoid = character and character:FindFirstChildWhichIsA('Humanoid')
		if not root or not humanoid or humanoid.Health <= 0 then lastSafe = nil return end
		local now = Safety.clock()
		if not Safety.deathArmed then
			if humanoid.Health >= 0.5 * humanoid.MaxHealth then
				Safety.healthySince = Safety.healthySince or now
				if now - Safety.healthySince >= 0.5 and now - Safety.lastDeathAt >= 2 then Safety.deathArmed = true end
			else
				Safety.healthySince = nil
			end
		end
		table.insert(Safety.history, { t = now, h = humanoid.Health, max = humanoid.MaxHealth })
		while #Safety.history > 0 and now - Safety.history[1].t > HISTORY_SECONDS do table.remove(Safety.history, 1) end
		if (Safety.rangeScale < 1 or Safety.jitterScale < 1) and now - Safety.lastRecover >= RECOVER_EVERY then
			Safety.lastRecover = now
			Safety.rangeScale = math.min(1, Safety.rangeScale * 1.25)
			Safety.jitterScale = math.min(1, Safety.jitterScale * 1.25)
		end
		if root.AssemblyLinearVelocity.Magnitude > MAX_SPEED then
			root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			Safety.flings += 1
		end
		if root.Position.Y < Workspace.FallenPartsDestroyHeight + FALL_MARGIN then
			if lastSafe then
				root.CFrame = lastSafe
				root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
				Safety.rescues += 1
			end
		elseif humanoid.FloorMaterial ~= Enum.Material.Air then
			lastSafe = root.CFrame
			local bounds = env.RivalsBounds
			if bounds then bounds:MarkMine(root) end
		end
	end

	RunService:BindToRenderStep('RivalsSafetyTrack', 120, track)
end

local oldBounds = env.RivalsBounds
if oldBounds and oldBounds.connection then oldBounds.connection:Disconnect() end
env.RivalsBounds = {
	cells = oldBounds and oldBounds.cells or {},
	connection = nil, lastMine = nil, lastMark = -math.huge,
}
do
	local Bounds = env.RivalsBounds
	local CELL = 12
	local KEEP = 120
	local BELOW = 4
	local ABOVE = 24
	local MINE_EVERY = 0.25
	local OTHERS_EVERY = 0.5
	local PRUNE_EVERY = 10
	local TELEPORT = 150
	local STAND_RAY = 6

	local function cellKey(cx, cz) return (cx + 4096) * 8192 + (cz + 4096) end

	function Bounds:Mark(position)
		local cx = math.floor(position.X / CELL)
		local cz = math.floor(position.Z / CELL)
		self.cells[cellKey(cx, cz)] = { y = position.Y, t = env.RivalsClock() }
	end

	function Bounds:Allows(position)
		local cx = math.floor(position.X / CELL)
		local cz = math.floor(position.Z / CELL)
		local now = env.RivalsClock()
		for dx = -1, 1 do
			for dz = -1, 1 do
				local cell = self.cells[cellKey(cx + dx, cz + dz)]
				if cell and now - cell.t <= KEEP and position.Y >= cell.y - BELOW and position.Y <= cell.y + ABOVE then return true end
			end
		end
		return false
	end

	function Bounds:MarkMine(root)
		local now = env.RivalsClock()
		if now - self.lastMark < MINE_EVERY then return end
		self.lastMark = now
		local position = root.Position
		if self.lastMine and (position - self.lastMine).Magnitude > TELEPORT then self.cells = {} end
		self.lastMine = position
		self:Mark(position)
	end

	function Bounds:Prune()
		local now = env.RivalsClock()
		for key, cell in pairs(self.cells) do
			if now - cell.t > KEEP then self.cells[key] = nil end
		end
	end

	function Bounds:Count()
		local count = 0
		for _ in pairs(self.cells) do count += 1 end
		return count
	end

	local standParams = RaycastParams.new()
	standParams.FilterType = Enum.RaycastFilterType.Exclude
	standParams.IgnoreWater = true

	local clock, pruneClock = 0, 0
	Bounds.connection = RunService.Heartbeat:Connect(function(dt)
		clock += dt or 0
		if clock < OTHERS_EVERY then return end
		pruneClock += clock
		clock = 0
		for _, other in next, Players:GetPlayers() do
			if other ~= LocalPlayer then
				local character = other.Character
				local root = character and character:FindFirstChild('HumanoidRootPart')
				local humanoid = character and character:FindFirstChildWhichIsA('Humanoid')
				if root and humanoid and humanoid.Health > 0 then
					standParams.FilterDescendantsInstances = { character }
					if Workspace:Raycast(root.Position, Vector3.new(0, -STAND_RAY, 0), standParams) then Bounds:Mark(root.Position) end
				end
			end
		end
		if pruneClock >= PRUNE_EVERY then pruneClock = 0 Bounds:Prune() end
	end)
end

local MAX_TARGET_DISTANCE = 150
local SANE_DISTANCE = 400
local GROUND_CHECK = 80
local CLEARANCE = 2.5

local HEAD_POINTS = {
	Vector3.new(0, 0.4, 0), Vector3.new(0.4, 0, 0), Vector3.new(-0.4, 0, 0), Vector3.new(0, 0, 0.4), Vector3.new(0, 0, -0.4), Vector3.new(0, -0.3, 0),
}

local function headPoints(head)
	local center = head.Position
	local points = { center }
	local size, cframe = head.Size, head.CFrame
	if not size or not cframe then return points end
	for _, factor in ipairs(HEAD_POINTS) do table.insert(points, center + cframe:VectorToWorldSpace(factor * size)) end
	return points
end

env.DesyncController = {}
do
	local groundParams = RaycastParams.new()
	groundParams.FilterType = Enum.RaycastFilterType.Exclude
	groundParams.IgnoreWater = true
	groundParams.RespectCanCollide = true
	local clearanceParams = OverlapParams.new()
	clearanceParams.FilterType = Enum.RaycastFilterType.Exclude
	local sightParams = RaycastParams.new()
	sightParams.FilterType = Enum.RaycastFilterType.Exclude
	sightParams.IgnoreWater = true
	sightParams.RespectCanCollide = true

	local function ownFilter(extra)
		local filter = {}
		if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
		local camera = currentCamera()
		if camera then table.insert(filter, camera) end
		if extra then table.insert(filter, extra) end
		return filter
	end

	local surfaceParams = RaycastParams.new()
	surfaceParams.FilterType = Enum.RaycastFilterType.Include
	surfaceParams.IgnoreWater = true
	local PROBES = {
		Vector3.new(1, 0, 0), Vector3.new(-1, 0, 0), Vector3.new(0, 1, 0), Vector3.new(0, -1, 0), Vector3.new(0, 0, 1), Vector3.new(0, 0, -1),
	}
	for x = -1, 1, 2 do
		for y = -1, 1, 2 do
			for z = -1, 1, 2 do table.insert(PROBES, Vector3.new(x, y, z).Unit) end
		end
	end

	local function touchesSurface(position, parts, from)
		surfaceParams.FilterDescendantsInstances = parts
		if Workspace:Raycast(from, position - from, surfaceParams) then return true end
		for _, direction in ipairs(PROBES) do
			if Workspace:Raycast(position, direction * CLEARANCE, surfaceParams) then return true end
		end
		return false
	end

	local function isZone(part)
		if part.Name == 'Barriers' or (part.Parent and part.Parent.Name == 'Barriers') then return true end
		return part:HasTag('OutOfBoundsPart') or part:HasTag('KillBrick')
	end

	function DesyncController:Sight(from, to, head)
		sightParams.FilterDescendantsInstances = ownFilter()
		local result = Workspace:Raycast(from, to - from, sightParams)
		return result == nil or result.Instance == head
	end

	function DesyncController:HasLine(position, head)
		if not head then return true end
		return self:Sight(position + Vector3.new(0, 1.5, 0), head.Position, head)
	end

	function DesyncController:Free(position, extra)
		clearanceParams.FilterDescendantsInstances = ownFilter(extra)
		local ok, parts = pcall(function() return Workspace:GetPartBoundsInRadius(position, CLEARANCE, clearanceParams) end)
		if ok and parts then
			for _, part in ipairs(parts) do
				if part.CanCollide then return false end
			end
		end
		return true
	end

	local function anchorFor(list, targetRoot, head)
		if list and list.anchor == 'flat' then return CFrame.new(targetRoot.Position) end
		if list and list.anchor == 'head' then
			local character = targetRoot.Parent
			local box = character and character:FindFirstChild('HitboxHead')
			local part = (box and box:IsA('BasePart')) and box or head
			return part and CFrame.new(part.Position) or nil
		end
		return targetRoot.CFrame
	end

	local function blockedSpot(self, position)
		local player = self.avoidBlock
		local state = env.RivalsTargetState
		if not player or not state then return false end
		return state:Blocked(player, position + Vector3.new(0, 1.75, 0), self.blockWeapon) ~= nil
	end

	local function approachList(targetRoot, rootPart)
		local toward = rootPart.Position - targetRoot.Position
		local flat = Vector3.new(toward.X, 0, toward.Z)
		if flat.Magnitude < 1 then return nil end
		local forward = flat.Unit
		local side = Vector3.new(-forward.Z, 0, forward.X)
		local list = { anchor = 'flat' }
		for _, distance in ipairs({ 4, 7, 10 }) do
			for _, lateral in ipairs({ 0, 3, -3 }) do
				for _, lift in ipairs({ 0, 2.5 }) do
					table.insert(list, CFrame.new(forward * distance + side * lateral + Vector3.new(0, lift, 0)))
				end
			end
		end
		return list
	end

	function DesyncController:Pick(targetRoot, head, rootPart)
		local now = env.RivalsClock()
		local cached = self.pick
		if cached and (cached.line or now - cached.at < 0.25) then
			local base = anchorFor(cached.list, targetRoot, head)
			local candidate = base and base * cached.offset
			if candidate then
				local safe, exact = self:IsSafe(candidate.Position, rootPart)
				if safe and (not cached.line or self:HasLine(candidate.Position, head)) and (cached.blocked or not blockedSpot(self, candidate.Position)) then
					self.pickExact = exact == true
					return candidate, cached.offset
				end
			end
		end
		local firstSafe, firstOffset, firstList, firstExact = nil, nil, nil, false
		local lastSafe, lastOffset, lastList, lastExact = nil, nil, nil, false
		for _, list in ipairs({ self.offsets or { self.offset or CFrame.new(0, 7, -2) }, self.fallback or {} }) do
			local base = anchorFor(list, targetRoot, head)
			if base then
				for _, offset in ipairs(list) do
					local candidate = base * offset
					local safe, exact = self:IsSafe(candidate.Position, rootPart)
					if safe then
						if blockedSpot(self, candidate.Position) then
							if not lastSafe then lastSafe, lastOffset, lastList, lastExact = candidate, offset, list, exact == true end
						elseif self:HasLine(candidate.Position, head) then
							self.pick = { offset = offset, list = list, line = head ~= nil, at = now }
							self.pickExact = exact == true
							return candidate, offset
						elseif not firstSafe then
							firstSafe, firstOffset, firstList, firstExact = candidate, offset, list, exact == true
						end
					end
				end
			end
		end
		local approach = v65('approach') and approachList(targetRoot, rootPart) or nil
		if approach then
			local base = anchorFor(approach, targetRoot, head)
			for _, offset in ipairs(approach) do
				local candidate = base * offset
				local safe, exact = self:IsSafe(candidate.Position, rootPart)
				if safe and not blockedSpot(self, candidate.Position) and self:HasLine(candidate.Position, head) then
					self.pick = { offset = offset, list = approach, line = head ~= nil, at = now }
					self.pickExact = exact == true
					self.approached = (self.approached or 0) + 1
					return candidate, offset
				end
			end
		end
		if firstSafe then
			self.pick = { offset = firstOffset, list = firstList, line = false, at = now }
			self.pickExact = firstExact
			return firstSafe, firstOffset
		end
		if lastSafe then
			self.pick = { offset = lastOffset, list = lastList, line = false, at = now, blocked = true }
			self.pickExact = lastExact
			return lastSafe, lastOffset
		end
		self.pick = nil
		self.pickExact = false
		return nil
	end

	function DesyncController:init()
		self.active = false
		self.connection = nil
		self.currentTarget = nil
	end

	function DesyncController:IsSafe(position, rootPart)
		local limit = MAX_TARGET_DISTANCE * env.RivalsSafety.rangeScale + 20
		if not ((position - rootPart.Position).Magnitude <= limit) then return false end
		if not (position.Y >= Workspace.FallenPartsDestroyHeight + 50) then return false end
		local filter = {}
		local camera = currentCamera()
		if camera then table.insert(filter, camera) end
		if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
		local target = self.currentTarget
		if target and target.Character then table.insert(filter, target.Character) end
		groundParams.FilterDescendantsInstances = filter
		clearanceParams.FilterDescendantsInstances = filter
		local ok, parts = pcall(function() return Workspace:GetPartBoundsInRadius(position, CLEARANCE, clearanceParams) end)
		local solid = {}
		if ok and parts then
			for _, part in ipairs(parts) do
				if isZone(part) then self.zoned = true return false end
				if part.CanCollide then table.insert(solid, part) end
			end
		end
		local exact = false
		if #solid > 0 then
			local character = target and target.Character
			local anchor = character and (character:FindFirstChild('Head') or character:FindFirstChild('HumanoidRootPart'))
			if not anchor or touchesSurface(position, solid, anchor.Position) then return false end
			exact = true
		end
		local ground = Workspace:Raycast(position, Vector3.new(0, -GROUND_CHECK, 0), groundParams)
		if ground == nil then return false end
		if isZone(ground.Instance) then self.zoned = true return false end
		return true, exact
	end

	local function faceToward(position, point)
		local flat = Vector3.new(point.X, position.Y, point.Z)
		if (flat - position).Magnitude < 0.05 then return CFrame.new(position) end
		return CFrame.lookAt(position, flat)
	end

	function DesyncController:Start(target)
		self:Stop(self.startTarget ~= target and 'switch' or 'restart')
		event('디싱크', target and target.Name or '?')
		self.active = true
		self.pick = nil
		self.placedAt = nil
		self.prewarmed = nil
		self.startTarget = target
		self.startedAt = env.RivalsClock()
		self.expireAt = self.startedAt + 0.6
		self.connection = RunService.Heartbeat:Connect(function()
			if not self.active then return end
			if self.pendingRestore then self.pendingRestore() end
			if env.RivalsClock() > (self.expireAt or math.huge) then self:Stop('expire') return end
			local rootPart = Character.RootPart
			if not rootPart then return end
			local targetRoot = target and target.Character and target.Character:FindFirstChild('HumanoidRootPart')
			if not targetRoot then self:Stop('lost') return end
			self.currentTarget = target
			local head = target.Character:FindFirstChild('Head')
			self.zoned = false
			local desyncCFrame, offset = self:Pick(targetRoot, head, rootPart)
			if self.zoned then env.RivalsSafety.zoneRefusals += 1 end
			if desyncCFrame then self.offset = offset end
			if not desyncCFrame then
				self.placed = nil
				self.placedAt = nil
				return
			end
			desyncCFrame = faceToward(desyncCFrame.Position, (head or targetRoot).Position)
			self.placed = desyncCFrame
			self.placedAt = self.placedAt or env.RivalsClock()
			self.placedExact = self.pickExact == true
			local backupCFrame = rootPart.CFrame
			local backupVelocity = rootPart.Velocity
			local backupRotVelocity = rootPart.RotVelocity
			rootPart.CFrame = desyncCFrame
			env.RivalsSafety.lastDesync = env.RivalsSafety.clock()
			local function restore()
				self.pendingRestore = nil
				rootPart.CFrame = backupCFrame
				rootPart.Velocity = backupVelocity
				rootPart.RotVelocity = backupRotVelocity
				pcall(function() RunService:UnbindFromRenderStep('desync_fallback') end)
			end
			if pcall(function() RunService:BindToRenderStep('desync_fallback', Enum.RenderPriority.First.Value, restore) end) then
				self.pendingRestore = restore
			else
				restore()
			end
		end)
	end

	function DesyncController:Stop(reason)
		if self.active and reason then
			event('디싱크 끝', string.format('%s %.2fs', reason, env.RivalsClock() - (self.startedAt or env.RivalsClock())))
		end
		self.active = false
		self.currentTarget = nil
		self.startTarget = nil
		self.placed = nil
		self.placedAt = nil
		self.pick = nil
		if self.pendingRestore then self.pendingRestore() end
		if self.connection then self.connection:Disconnect() self.connection = nil end
	end

	DesyncController:init()
end

env.RivalsTargetState = {
	flagged = setmetatable({}, { __mode = 'k' }),
	cache = setmetatable({}, { __mode = 'k' }),
	disprovals = 0, probes = 0, probeHits = 0,
	blocked = 0, flankShots = 0, flankHits = 0, controlShots = 0, controlHits = 0,
}
do
	local TargetState = env.RivalsTargetState
	local FRONT_BLOCK_DOT = -0.3778
	local BACK_BLOCK_DOT = 0.70355
	local DEFLECT_BLOCK_DOT = -0.7136558449999999
	local SHIELD_IMMUNE = { ['Flare Gun'] = true, ['Grenade Launcher'] = true, RPG = true, Exogun = true, Distortion = true, Flamethrower = true, Spray = true }
	local DEFLECT_IMMUNE = { ['Flare Gun'] = true, ['Grenade Launcher'] = true, RPG = true, Exogun = true, Distortion = true, Flamethrower = true, Spray = true, Daggers = true, Bow = true, Crossbow = true, Slingshot = true }
	local CACHE_FOR = 0.05
	local PROBE_EVERY = 3
	local PROBE_FOR = 0.15
	local DISPROVE_AT = 2
	local CONTROL_MIN = 25
	local CONTROL_MAX_RATE = 0.12
	TargetState.BLOCK_CONTROL = 0.15

	local fighterController
	pcall(function()
		local module = PlayerScripts.Controllers:FindFirstChild('FighterController')
		if module then fighterController = require(module) end
	end)
	note('FighterController 모듈', fighterController ~= nil)

	local function fighterOf(player)
		if fighterController == nil then return nil end
		local fighter
		pcall(function() fighter = fighterController:GetFighter(player) end)
		if type(fighter) ~= 'table' then
			local map = type(fighterController) == 'table' and rawget(fighterController, '_player_to_fighter') or nil
			fighter = type(map) == 'table' and map[player] or nil
		end
		return type(fighter) == 'table' and fighter or nil
	end

	local function field(object, key)
		local value = rawget(object, key)
		if value == nil then pcall(function() value = object[key] end) end
		return value
	end

	local function isVector2(value)
		local ok, x, y = pcall(function() return value.X, value.Y end)
		return ok and type(x) == 'number' and type(y) == 'number'
	end

	local function lookVector(rotation)
		local pitch, yaw = rotation.X, rotation.Y
		local flat = math.cos(pitch)
		return Vector3.new(-math.sin(yaw) * flat, math.sin(pitch), -math.cos(yaw) * flat)
	end

	local function gameClock()
		if type(tick) == 'function' then return tick() end
		return os.time()
	end

	local function readState(player)
		local fighter = fighterOf(player)
		if not fighter then return nil end
		local state = { invincible = false, crouching = false }
		local entity = field(fighter, 'Entity')
		local entityData = type(entity) == 'table' and field(entity, 'Data') or nil
		state.invincible = type(entityData) == 'table' and field(entityData, 'IsInvincible') == true
		local data = field(fighter, 'Data')
		if type(data) == 'table' then
			state.crouching = field(data, 'IsCrouching') == true
			local rotation = field(data, 'CameraRotation')
			if rotation ~= nil and isVector2(rotation) then state.look = lookVector(rotation) end		end
		local equipped = field(fighter, 'EquippedItem')
		local riotEquipped = false
		if type(equipped) == 'table' then
			local info = field(equipped, 'Info')
			if type(info) == 'table' and info.Type == 'Melee' then
				local now = gameClock()
				state.melee = true
				riotEquipped = field(equipped, 'Name') == 'Riot Shield'
				local deflectEnd = field(equipped, '_deflect_cooldown')
				local deflectCooldown = info.DeflectCooldown
				state.deflecting = type(deflectEnd) == 'number' and type(deflectCooldown) == 'number' and now < deflectEnd - deflectCooldown
				local attackEnd = field(equipped, '_attack_cooldown')
				local attacking = type(attackEnd) == 'number' and now < attackEnd
				state.riotFront = riotEquipped and not attacking
			end
		end
		if not riotEquipped then
			local items = field(fighter, 'Items')
			if type(items) == 'table' then
				for _, item in pairs(items) do
					if type(item) == 'table' and field(item, 'Name') == 'Riot Shield' then state.riotBack = true break end
				end
			end
		end
		return state
	end

	function TargetState:Read(player)
		local now = env.RivalsClock()
		local cached = self.cache[player]
		if cached and now - cached.at < CACHE_FOR then return cached.state end
		local ok, state = pcall(readState, player)
		if not ok then state = nil end
		self.cache[player] = { at = now, state = state }
		return state
	end

	function TargetState:IsInvincible(player)
		local state = self:Read(player)
		return state ~= nil and state.invincible == true
	end

	function TargetState:HoldsMelee(player)
		local state = self:Read(player)
		return state ~= nil and state.melee == true
	end

	function TargetState:Unshootable(player)
		if self.disproved or not self:IsInvincible(player) then
			self.flagged[player] = nil
			return false
		end
		local now = env.RivalsClock()
		local since = self.flagged[player]
		if not since then since = now self.flagged[player] = since end
		local elapsed = now - since
		return not (elapsed >= PROBE_EVERY and elapsed % PROBE_EVERY < PROBE_FOR)
	end

	function TargetState:Disprove()
		self.disprovals += 1
		if self.disprovals >= DISPROVE_AT and not self.disproved then
			self.disproved = true
			warn('[YGJ] 무적 판독이 틀렸다. 판독 오프')
		end
	end

	function TargetState:Blocked(player, origin, weaponName)
		if self.blockOff then return nil end
		local state = self:Read(player)
		if not state or not state.look then return nil end
		local character = player.Character
		local root = character and character:FindFirstChild('HumanoidRootPart')
		if not root then return nil end
		local center = root.Position + Vector3.new(0, state.crouching and 0.25 or 1.75, 0)
		local offset = center - origin
		local distance = offset.Magnitude
		local dot = distance > 1e-4 and offset:Dot(state.look) / distance or 0
		if state.deflecting and not DEFLECT_IMMUNE[weaponName] and dot < DEFLECT_BLOCK_DOT then return 'deflect' end
		if state.riotFront and not SHIELD_IMMUNE[weaponName] and dot < FRONT_BLOCK_DOT then return 'front' end
		if state.riotBack and not SHIELD_IMMUNE[weaponName] and dot > BACK_BLOCK_DOT then return 'back' end
		return nil
	end

	function TargetState:LookDot(player, origin)
		local state = self:Read(player)
		if not state or not state.look or not origin then return nil end
		local character = player.Character
		local root = character and character:FindFirstChild('HumanoidRootPart')
		if not root then return nil end
		local offset = root.Position + Vector3.new(0, state.crouching and 0.25 or 1.75, 0) - origin
		local distance = offset.Magnitude
		if distance < 1e-4 then return 0 end
		return offset:Dot(state.look) / distance
	end

	function TargetState:Equipped(player)
		local fighter = fighterOf(player)
		if not fighter then return nil end
		local item = field(fighter, 'EquippedItem')
		return type(item) == 'table' and item or nil
	end

	function TargetState:Crouching(player)
		local state = self:Read(player)
		if not state then return nil end
		return state.crouching == true
	end

	function TargetState:Settle(info, hit)
		if info.invincible then
			self.probes += 1
			if hit then self.probeHits += 1 self:Disprove() end
		end
		if info.flank then
			self.flankShots += 1
			if hit then self.flankHits += 1 end
		end
		if info.control then
			self.controlShots += 1
			if hit then self.controlHits += 1 end
			if not self.blockOff and self.controlShots >= CONTROL_MIN and self.controlHits / self.controlShots > CONTROL_MAX_RATE then
				self.blockOff = true
				warn('[YGJ] 방패 판독이 틀렸다. 판독 오프')
			end
		end
	end

	function TargetState:Summary()
		if self.probes == 0 and self.blocked == 0 and self.disprovals == 0 then return nil end
		return string.format('판독: 무적 시험 %d발(맞음 %d)%s  막힘 %d (우회 %d/%d, 대조 %d/%d)%s',
			self.probes, self.probeHits, self.disproved and ' [무적 판독 끔]' or '',
			self.blocked, self.flankHits, self.flankShots, self.controlHits, self.controlShots, self.blockOff and ' [방패 판독 끔]' or '')
	end
end

env.TargetController = {}
do
	function TargetController:init()
		self.active = true
		self.target = nil
		self.connection = RunService.Heartbeat:Connect(function()
			if not self.active then return end
			local before = self.target
			self.target = self:GetClosestTarget()
			if self.target ~= before then
				event('대상', self.target and string.format('%s (%s)%s', self.target.Name, self.targetVisible and '보임' or (self.targetInRange and '벽 뒤' or '먼 벽 뒤'), self.meleeFirst and ', 근접 무기라 먼저' or '') or '없음')
			end
		end)
	end

	function TargetController:IsValidTarget(character)
		local rootPart = character:FindFirstChild('HumanoidRootPart')
		local head = character:FindFirstChild('Head')
		local humanoid = character:FindFirstChildWhichIsA('Humanoid')
		return rootPart and head and humanoid and humanoid.Health > 0 or false
	end

	function TargetController:IsValidTeam(player)
		return player:GetAttribute('TeamID') ~= LocalPlayer:GetAttribute('TeamID')
	end

	function TargetController:IsValidMatch(player)
		return player:GetAttribute('EnvironmentID') == LocalPlayer:GetAttribute('EnvironmentID')
	end

	function TargetController:IsValidCandidate(player, ignoreRange)
		if player == LocalPlayer then return false end
		if not self:IsValidTeam(player) then return false end
		if not self:IsValidMatch(player) then return false end
		local character = player.Character
		if not character then return false end
		if not self:IsValidTarget(character) then return false end
		if character:FindFirstChildWhichIsA('ForceField') then return false end
		local camera = currentCamera()
		if not camera then return false end
		local head = character:FindFirstChild('Head')
		local span = (head.Position - camera.CFrame.Position).Magnitude
		if v65('sanity') and (span ~= span or span > SANE_DISTANCE) then return false end
		if ignoreRange then return true end
		if span > MAX_TARGET_DISTANCE * env.RivalsSafety.rangeScale then return false end
		return true
	end

	function TargetController:VisibleEnemy()
		local camera = currentCamera()
		if not camera then return nil end
		local cameraPosition = camera.CFrame.Position
		local best, bestPoint, bestDistance = nil, nil, math.huge
		for _, player in next, Players:GetPlayers() do
			if self:IsShootable(player, true) then
				local visible, _, distance, point = self:Judge(player, cameraPosition)
				if visible and distance < bestDistance then best, bestPoint, bestDistance = player, point, distance end
			end
		end
		return best, bestPoint, bestDistance
	end

	local viewParams = RaycastParams.new()
	viewParams.FilterType = Enum.RaycastFilterType.Exclude
	viewParams.IgnoreWater = true
	viewParams.RespectCanCollide = true

	local BODY_PARTS = { 'HitboxBody', 'UpperTorso', 'Torso', 'HumanoidRootPart' }
	local BODY_PARTS_SMALL = { 'HitboxBodySmall', 'HitboxBody', 'UpperTorso', 'Torso', 'HumanoidRootPart' }

	function TargetController:VisibleBody(player, cameraPosition, weaponName)
		local character = player.Character
		if not character then return nil end
		local part = nil
		for _, name in ipairs(weaponName == 'Sniper' and BODY_PARTS_SMALL or BODY_PARTS) do
			local candidate = character:FindFirstChild(name)
			if candidate and candidate:IsA('BasePart') then part = candidate break end
		end
		if not part then return nil end
		local filter = { character }
		if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
		local camera = currentCamera()
		if camera then table.insert(filter, camera) end
		viewParams.FilterDescendantsInstances = filter
		for _, point in ipairs(headPoints(part)) do
			if Workspace:Raycast(cameraPosition, point - cameraPosition, viewParams) == nil then return part, point end
		end
		return nil
	end

	function TargetController:VisibleBodyEnemy(weaponName)
		local camera = currentCamera()
		if not camera then return nil end
		local cameraPosition = camera.CFrame.Position
		local best, bestPart, bestPoint, bestDistance = nil, nil, nil, math.huge
		for _, player in next, Players:GetPlayers() do
			if self:IsShootable(player, false) then
				local part, point = self:VisibleBody(player, cameraPosition, weaponName)
				if part then
					local distance = (point - cameraPosition).Magnitude
					if distance < bestDistance then best, bestPart, bestPoint, bestDistance = player, part, point, distance end
				end
			end
		end
		return best, bestPart, bestPoint
	end

	function TargetController:IsShootable(player, ignoreRange)
		if not self:IsValidCandidate(player, ignoreRange) then return false end
		return not env.RivalsTargetState:Unshootable(player)
	end

	local LOCK_STALE = 2
	local LOCK_CLOSER = 0.5
	local HIDDEN_SWITCH = 0.3
	local MELEE_FIRST = 18

	function TargetController:VisiblePoint(player, cameraPosition)
		local character = player.Character
		local head = character and character:FindFirstChild('Head')
		if not head then return nil end
		local filter = { character }
		if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
		local camera = currentCamera()
		if camera then table.insert(filter, camera) end
		viewParams.FilterDescendantsInstances = filter
		for _, point in ipairs(headPoints(head)) do
			if Workspace:Raycast(cameraPosition, point - cameraPosition, viewParams) == nil then return point end
		end
		return nil
	end

	function TargetController:IsVisible(player, cameraPosition)
		return self:VisiblePoint(player, cameraPosition) ~= nil
	end

	function TargetController:Judge(player, cameraPosition)
		if not cameraPosition then
			local camera = currentCamera()
			if not camera then return false, false, math.huge end
			cameraPosition = camera.CFrame.Position
		end
		local head = player.Character and player.Character:FindFirstChild('Head')
		if not head then return false, false, math.huge end
		local distance = (head.Position - cameraPosition).Magnitude
		local point = self:VisiblePoint(player, cameraPosition)
		local inRange = distance <= MAX_TARGET_DISTANCE * env.RivalsSafety.rangeScale
		return point ~= nil, inRange, distance, point
	end

	function TargetController:GetClosestTarget()
		local camera = currentCamera()
		if not camera then self.targetVisible, self.targetInRange = false, false return nil end
		local cameraPosition = camera.CFrame.Position
		local judged = {}
		local function judge(player)
			local entry = judged[player]
			if not entry then entry = { self:Judge(player, cameraPosition) } judged[player] = entry end
			return entry[1], entry[2], entry[3]
		end
		local current = self.target
		local now = env.RivalsClock()
		self.meleeFirst = false

		local threat = self:NearestMeleeThreat(MELEE_FIRST)
		if threat and threat ~= current and self:IsShootable(threat, true) then
			local threatVisible, threatInRange = judge(threat)
			if threatVisible then
				self.hiddenSince = nil
				self.meleeFirst = true
				self.targetVisible, self.targetInRange = true, threatInRange
				return threat
			end
		end

		local valid = current ~= nil and self:IsShootable(current, true)
		local currentVisible, currentInRange, currentDistance = false, false, math.huge
		if valid then
			currentVisible, currentInRange, currentDistance = judge(current)
			if not currentVisible and not currentInRange then valid = false end
		end
		if valid then
			if self.lockTarget ~= current then self.lockTarget, self.lockAt = current, now end
			if currentVisible then self.hiddenSince = nil else self.hiddenSince = self.hiddenSince or now end
			local since = self.lockAt
			local stats = env.RivalsStats
			local lastHit = stats and stats.lastHit
			if lastHit and lastHit.target == current and lastHit.t > since then since = lastHit.t end
			local stale = now - since >= LOCK_STALE
			local hiddenLong = not currentVisible and now - self.hiddenSince >= HIDDEN_SWITCH
			if not stale and not hiddenLong and (currentVisible or currentInRange) then
				self.targetVisible, self.targetInRange = currentVisible, currentInRange
				return current
			end
		else
			self.lockTarget = nil
			self.hiddenSince = nil
		end

		local best, bestVisible, bestInRange, bestDistance = nil, false, false, math.huge
		local bestRank = math.huge
		for _, player in next, Players:GetPlayers() do
			if self:IsShootable(player, true) then
				local visible, inRange, distance = judge(player)
				local rank = visible and 1 or (inRange and 2 or nil)
				if rank and (rank < bestRank or (rank == bestRank and distance < bestDistance)) then
					best, bestVisible, bestInRange, bestDistance, bestRank = player, visible, inRange, distance, rank
				end
			end
		end

		local pick = best
		if valid then
			pick = current
			local stale = now - (self.lockAt or now) >= LOCK_STALE
			local stats = env.RivalsStats
			local lastHit = stats and stats.lastHit
			if lastHit and lastHit.target == current and now - lastHit.t < LOCK_STALE then stale = false end
			if best ~= nil and best ~= current then
				if bestVisible and not currentVisible then pick = best
				elseif bestInRange and not currentInRange and not currentVisible then pick = best
				elseif stale and bestVisible == currentVisible and bestDistance < LOCK_CLOSER * currentDistance then pick = best end
			end
		end
		if pick == current and valid then
			self.targetVisible, self.targetInRange = currentVisible, currentInRange
		elseif pick then
			self.targetVisible, self.targetInRange = bestVisible, bestInRange
		else
			self.targetVisible, self.targetInRange = false, false
		end
		return pick
	end

	function TargetController:NearestEnemy(range, meleeOnly)
		local root = Character.RootPart
		if not root then return nil end
		local limit = range or math.huge
		local targetState = env.RivalsTargetState
		local best, bestDistance, bestRoot = nil, nil, nil
		for _, player in next, Players:GetPlayers() do
			if self:IsValidCandidate(player, true) then
				local enemyRoot = player.Character:FindFirstChild('HumanoidRootPart')
				if enemyRoot then
					local distance = (enemyRoot.Position - root.Position).Magnitude
					if distance <= limit and (not best or distance < bestDistance) and (not meleeOnly or targetState:HoldsMelee(player)) then
						best, bestDistance, bestRoot = player, distance, enemyRoot
					end
				end
			end
		end
		if not best then return nil end
		return best, bestDistance, bestRoot
	end

	function TargetController:NearestEnemyDistance()
		local _, distance = self:NearestEnemy()
		return distance
	end

	function TargetController:NearestMeleeThreat(range)
		return self:NearestEnemy(range, true)
	end

	function TargetController:Stop()
		self.active = false
		if self.connection then self.connection:Disconnect() self.connection = nil end
	end

	TargetController:init()
end

local CTX_DIMS = { point = true, hitpart = true }
local STILL_SPEED = 1

local previousTuner = env.RivalsTuner
env.RivalsTuner = { dims = {}, random = math.random, credits = 0, version = 6 }
do
	local Tuner = env.RivalsTuner
	local MIN_TRIALS = 5
	local EXPLORE = 0.05
	local EXPLORE_SURE = 0.02
	local SURE_AFTER = 15
	local SURE_RATIO = 2.5
	local DECAY_AT = 300

	function Tuner:AddDimension(name, arms)
		local dim = { order = arms, arms = {} }
		for _, arm in ipairs(arms) do dim.arms[arm] = { assigned = 0, finished = 0, hits = 0 } end
		self.dims[name] = dim
	end

	local function score(arm) return (arm.hits + 1) / (arm.finished + 2) end

	function Tuner:BestOther(name, weapon, exclude)
		if weapon then name = self:Scoped(name, weapon) end
		local dim = self.dims[name]
		if not dim then return nil end
		local best, bestScore = nil, -1
		for _, arm in ipairs(dim.order) do
			if arm ~= exclude then
				local s = score(dim.arms[arm])
				if s > bestScore then best, bestScore = arm, s end
			end
		end
		if best then dim.arms[best].assigned += 1 end
		return best
	end

	function Tuner:Best(name)
		local dim = self.dims[name]
		local best, bestScore = dim.order[1], -1
		for _, arm in ipairs(dim.order) do
			local s = score(dim.arms[arm])
			if s > bestScore then best, bestScore = arm, s end
		end
		return best
	end

	function Tuner:Choose(name, weapon)
		if weapon then name = self:Scoped(name, weapon) end
		local dim = self.dims[name]
		local pick, fewest = nil, math.huge
		local mode = 'trial'
		for _, arm in ipairs(dim.order) do
			local assigned = dim.arms[arm].assigned
			if assigned < MIN_TRIALS and assigned < fewest and dim.arms[arm].finished < 3 * MIN_TRIALS then pick, fewest = arm, assigned end
		end
		if not pick then
			local best = self:Best(name)
			local explore = EXPLORE
			if dim.arms[best].finished >= SURE_AFTER then
				local rival = 0
				for _, armName in ipairs(dim.order) do
					if armName ~= best then rival = math.max(rival, score(dim.arms[armName])) end
				end
				if score(dim.arms[best]) >= SURE_RATIO * rival then explore = EXPLORE_SURE end
			end
			if self.random() < explore then
				local index = math.min(#dim.order, math.floor(self.random() * #dim.order) + 1)
				pick = dim.order[index]
				mode = 'explore'
			else
				pick = best
				mode = 'best'
			end
		end
		dim.arms[pick].assigned += 1
		return pick, mode
	end

	function Tuner:Usable(name)
		local dim = self.dims[name]
		if not dim then return false end
		local finished = 0
		for _, arm in pairs(dim.arms) do finished += arm.finished end
		if finished < 20 or score(dim.arms[self:Best(name)]) >= 0.12 then return true end
		local now = env.RivalsClock()
		if not self.probeAt then self.probeAt = now - 1
		elseif now - self.probeAt >= 30 then self.probeAt = now end
		return now - self.probeAt < 1
	end

	function Tuner:Penalize(name, armName, weight)
		local dim = self.dims[name]
		local arm = dim and dim.arms[armName]
		if not arm then return end
		arm.finished += weight or 1
	end

	function Tuner:Credit(name, armName, hit)
		local dim = self.dims[name]
		local arm = dim and dim.arms[armName]
		if not arm then return end
		arm.finished += 1
		if hit then arm.hits += (type(hit) == 'number' and hit or 1) end
		self.credits += 1
		if self.credits % 20 == 0 and env.RivalsClock() - (self.savedAt or -math.huge) >= 5 then
			self.savedAt = env.RivalsClock()
			if self.Save then self:Save() end
		end
		local total = 0
		for _, a in pairs(dim.arms) do total += a.finished end
		if total > DECAY_AT then
			for _, a in pairs(dim.arms) do
				a.finished = a.finished / 2
				a.hits = a.hits / 2
				a.assigned = math.min(a.assigned, MIN_TRIALS)
			end
		end
	end

	Tuner:AddDimension('origin', { 'under', 'shooter', 'clear' })
	Tuner:AddDimension('desync', { 'above', 'glue' })
	Tuner:AddDimension('point', { 'center', 'lead', 'split', 'ahead' })
	Tuner:AddDimension('facing', { 'look', 'old' })
	Tuner:AddDimension('direct', { 'near', 'camera' })
	Tuner:AddDimension('hitpart', { 'hitbox', 'head', 'box' })
	Tuner:AddDimension('reach', { 'move', 'stay' })
	Tuner:AddDimension('reachNear', { 'move', 'stay' })
	Tuner:AddDimension('flankReach', { 'near', 'mid', 'far' })
	Tuner:AddDimension('meleeStill', { 'far', 'near' })

	local SEEDS = {}
	local SEED_CAP = 150
	local PRIOR = 0.2

	function Tuner:Scoped(name, weapon)
		local key = name .. '@' .. weapon
		if self.dims[key] then return key end
		local base = self.dims[name]
		if not base then return name end
		local seed = SEEDS[key]
		local total = 0
		if seed then
			for _, armName in ipairs(base.order) do if seed[armName] then total += seed[armName][2] end end
		end
		local scale = total > SEED_CAP and SEED_CAP / total or 1
		local dim = { order = base.order, arms = {} }
		for _, armName in ipairs(base.order) do
			local global, mine = base.arms[armName], seed and seed[armName]
			dim.arms[armName] = {
				assigned = MIN_TRIALS,
				finished = global.finished * PRIOR + (mine and mine[2] * scale or 0),
				hits = math.min(global.hits, global.finished) * PRIOR + (mine and mine[1] * scale or 0),
			}
		end
		self.dims[key] = dim
		return key
	end

	local FILE = 'ygj_tuner.json'
	local VERSION = 8

	local function inherit(source)
		for name in pairs(source) do
			local base, weapon = string.match(name, '^(.-)@(.+)$')
			if base then Tuner:Scoped(base, weapon) end
		end
		for name, dim in pairs(Tuner.dims) do
			local old = source[name]
			if type(old) == 'table' then
				for _, armName in ipairs(dim.order) do
					local arm, before = dim.arms[armName], old[armName]
					if type(before) == 'table' and type(before.finished) == 'number' and type(before.hits) == 'number' and before.finished >= 0 then
						arm.finished = before.finished * 0.5
						arm.hits = math.min(before.hits, before.finished) * 0.5
						arm.assigned = math.min(MIN_TRIALS, math.floor(arm.finished))
					end
				end
			end
		end
	end

	function Tuner:Save()
		if type(writefile) ~= 'function' then return false end
		local data = { version = VERSION, dims = {} }
		for name, dim in pairs(self.dims) do
			data.dims[name] = {}
			for armName, arm in pairs(dim.arms) do
				data.dims[name][armName] = { finished = arm.finished, hits = arm.hits }
			end
		end
		return (pcall(function() writefile(FILE, game:GetService('HttpService'):JSONEncode(data)) end))
	end

	function Tuner:LoadFile()
		if type(readfile) ~= 'function' or type(isfile) ~= 'function' then return false end
		local ok, data = pcall(function()
			if not isfile(FILE) then return nil end
			return game:GetService('HttpService'):JSONDecode(readfile(FILE))
		end)
		if ok and type(data) == 'table' and data.version == VERSION and type(data.dims) == 'table' then
			inherit(data.dims)
			return true
		end
		return false
	end

	if previousTuner and previousTuner.dims and previousTuner.version == VERSION then
		local previous = {}
		for name, dim in pairs(previousTuner.dims) do previous[name] = dim.arms end
		inherit(previous)
		Tuner.source = 'memory'
	elseif Tuner:LoadFile() then
		Tuner.source = 'file'
	end
end

	local function middle(sorted)
		local n = #sorted
		if n % 2 == 1 then return sorted[(n + 1) / 2] end
		return (sorted[n / 2] + sorted[n / 2 + 1]) / 2
	end

	function Stats:Sample()
		local controller = env.TargetController
		local player = controller and controller.target
		local character = player and player.Character
		local root = character and character:FindFirstChild('HumanoidRootPart')
		if not root then return end
		local now = self.clock()
		local track = self.track[player]
		if not track then
			self.track[player] = { pos = root.Position, t = now, started = now }
			return
		end
		local dt = now - track.t
		if dt < 0.1 then return end
		local speed = (root.Position - track.pos).Magnitude / dt
		track.speed = dt <= 0.5 and speed or nil
		if speed > 0.5 then track.movedAt = now end
		track.pos, track.t = root.Position, now
	end

	function Stats:Record(targetPlayer, info)
		local character = targetPlayer.Character
		local humanoid = character and character:FindFirstChildWhichIsA('Humanoid')
		info = info or {}

		local head = character and character:FindFirstChild('Head')
		local camera = currentCamera()
		if head and camera then info.distance = (head.Position - camera.CFrame.Position).Magnitude end
		local root = character and character:FindFirstChild('HumanoidRootPart')
		if root and root.AssemblyLinearVelocity then info.speed = root.AssemblyLinearVelocity.Magnitude end
		info.ping = self:Ping()
		if v65('log') then
			pcall(function()
				local x = info.x or {}
				info.x = x
				local track = self.track[targetPlayer]
				if track and self.clock() - track.t <= 0.5 then
					x.pv = track.speed
					x.sf = math.min(99, self.clock() - (track.movedAt or track.started))
				end
				local state = humanoid and humanoid:GetState()
				x.st = state and state.Name or nil
				x.md = humanoid and humanoid.MoveDirection.Magnitude or nil
				x.hp = humanoid and humanoid.Health or nil
				if head and root then x.hh = head.Position.Y - root.Position.Y end
			end)
		end

		local record = {
			target = targetPlayer, humanoid = humanoid,
			health = humanoid and humanoid.Health or 0,
			t = self.clock(), info = info, result = 'pending', weapon = info.weapon,
		}
		local window = info.window
		if window and not window.recorded then
			window.recorded = true
			info.lead = true
			self.windows += 1
			local list = self.targetWindows[targetPlayer]
			if not list then list = {} self.targetWindows[targetPlayer] = list end
			table.insert(list, { t = record.t, choices = window.choices })
			if #list > 12 then table.remove(list, 1) end
		end
		self.shots += 1
		table.insert(self.pending, record)
		table.insert(self.log, record)
		if #self.log > LOG_SIZE then table.remove(self.log, 1) end
		return record
	end

	function Stats:Resolve(record, hit, damage)
		record.result = hit and 'hit' or 'miss'
		local credit = hit
		if hit then
			self.hits += 1
			self.lastHit = { target = record.target, t = env.RivalsClock() }
			if record.target then self.lastHitAt[record.target] = math.max(self.lastHitAt[record.target] or -math.huge, record.t) end
			record.damage = damage
			record.delay = self.clock() - record.t
			if damage and record.weapon ~= nil then
				local best = math.max(self.maxDamage[record.weapon] or 0, damage)
				self.maxDamage[record.weapon] = best
				record.best = best
				local killed = record.humanoid ~= nil and record.humanoid.Health <= 0
				local ratio = best > 0 and math.min(1, damage / best) or 1
				self.weighedHits += 1
				if killed or ratio >= 0.9 then self.topHits += 1 end
				if killed then event('킬', record.target and record.target.Name or '?') end
				credit = killed and 1 or math.max(0.1, ratio * ratio)
			end
			local gun = record.info and record.info.weapon
			local crit = type(gun) == 'table' and tonumber(gun.CriticalDamage) or nil
			local body = type(gun) == 'table' and tonumber(gun.ShootDamage) or nil
			local isHead = damage and crit and crit > 0 and math.abs(damage - crit) <= 0.05 * crit
			local isBody = damage and crit and body and body > 0 and body < crit and math.abs(damage - body) <= 0.05 * body
			if isHead or isBody then
				local head = isHead and true or false
				self.headTotal += 1
				if head then self.headHits += 1 end
				local name = record.info.weaponName
				if name then
					local entry = self.headBy[name]
					if not entry then entry = { 0, 0 } self.headBy[name] = entry end
					entry[2] += 1
					if head then entry[1] += 1 end
				end
			end
		else
			self.misses += 1
		end
		local counted = true
		local window = record.info.window
		if window then
			if window.credited then counted = false
			else
				window.credited = true
				if window.dead then window.dead = nil self.deadWindows -= 1 end
				if hit then self.windowHits += 1 else self.windowMisses += 1 end
			end
		end
		if counted and record.info.arms and not record.info.invincible and not record.info.scanned and record.info.moved ~= false then
			local weaponName = record.info.weaponName
			local ctx = window and window.ctx
			local modes = window and window.modes
			for dimension, arm in pairs(record.info.arms) do
				if not (modes and modes[dimension] == 'switch') then
					env.RivalsTuner:Credit(dimension, arm, credit)
					if weaponName then
						local scoped = dimension .. '@' .. weaponName
						if ctx and CTX_DIMS[dimension] then scoped = scoped .. ctx end
						env.RivalsTuner:Credit(scoped, arm, credit)
					end
				end
			end
		end
		if env.RivalsTargetState then env.RivalsTargetState:Settle(record.info, hit) end
	end

	function Stats:MissRun(target)
		local list = self.targetWindows[target]
		if not list then return nil end
		local now = self.clock()
		local wait = math.max(0.3, (self:Ping() or 100) / 1000 + 0.15)
		local lastHit = self.lastHitAt[target] or -math.huge
		local missed, latest = 0, nil
		for i = #list, 1, -1 do
			local entry = list[i]
			if entry.t <= lastHit or now - entry.t > 3 then break end
			if now - entry.t >= wait then
				missed += 1
				latest = latest or entry
			end
		end
		if missed >= 2 then return latest.choices end
		return nil
	end

		if shape then
			if self.shapes[shape] == nil then

				else self.shapeCount += 1 end
			end
			self.shapes[shape] = (self.shapes[shape] or 0) + 1
		end
	end

	function Stats:Update()
		local now = self.clock()
		local newHealth = {}
		local keep = {}
		local credit = {}
		local latency = nil
		for _, record in ipairs(self.pending) do
			local humanoid = record.humanoid
			if humanoid ~= nil and humanoid.Health < record.health - 0.001 then
				latency = latency or ((self:Ping() or 100) / 1000)
				local gap = math.abs((now - record.t) - latency)
				local best = credit[humanoid]
				if not best or gap < best.gap then credit[humanoid] = { record = record, gap = gap, damage = record.health - humanoid.Health } end
			end
		end
		for _, record in ipairs(self.pending) do
			local humanoid = record.humanoid
			local best = humanoid ~= nil and credit[humanoid] or nil
			if best and best.record == record then
				newHealth[humanoid] = humanoid.Health
				self:Resolve(record, true, best.damage)
			elseif humanoid ~= nil and humanoid.Health <= 0 then
				record.result = 'dead'
				local window = record.info and record.info.window
				if window and not window.credited and not window.dead then window.dead = true self.deadWindows += 1 end
			elseif humanoid == nil or now - record.t > HIT_WINDOW then
				self:Resolve(record, false)
				if humanoid ~= nil then self.lateWatch[humanoid] = { record = record, health = humanoid.Health } end
			else
				table.insert(keep, record)
			end
		end
		local aimed = {}
		for _, record in ipairs(keep) do
			if record.humanoid then
				aimed[record.humanoid] = true
				if newHealth[record.humanoid] ~= nil then record.health = newHealth[record.humanoid] end
			end
		end
		self.pending = keep
		for humanoid, watch in pairs(self.lateWatch) do
			local health = humanoid.Health
			if now - watch.record.t > LATE_WINDOW then self.lateWatch[humanoid] = nil
			elseif aimed[humanoid] or credit[humanoid] or health >= watch.health - 0.001 then watch.health = health
			else
				watch.record.late = now - watch.record.t
				watch.record.lateDamage = watch.health - math.max(health, 0)
				self.lateWatch[humanoid] = nil
			end
		end
	end

	local function describeKey(key)
		if type(key) == 'string' then
			local bytes = {}
			for i = 1, math.min(#key, 8) do table.insert(bytes, tostring(string.byte(key, i))) end
			return 'str[' .. table.concat(bytes, ',') .. ']'
		end
		return tostring(key)
	end

	function Stats:CheckPayload(cameraData)
		if self.payloadChecked then return end
		self.payloadChecked = true
		local present = 0
		for i = 0, 3 do if cameraData[utf8.char(i)] ~= nil then present += 1 end end
		self.drift = present < 4
		if not self.drift then return end
		local keys = {}
		for key, value in pairs(cameraData) do
			if #keys >= 12 then break end
			table.insert(keys, describeKey(key) .. '=' .. typeof(value))
		end
		warn('[YGJ] 샷 데이터에 예상 키 0~3이 ' .. present .. '/4개만 있다. 실제 키: ' .. table.concat(keys, ', '))
	end

	local function fmt(number) return number and string.format('%.1f', number) or '-' end

	local function windowRows(log)
		local rows, byId = {}, {}
		for _, record in ipairs(log) do
			local info = record.info or {}
			local window = info.window
			if type(window) == 'table' and window.id ~= nil then
				local row = byId[window.id]
				if not row then
					row = { id = window.id, t = record.t, record = record, info = info, requests = 0, damage = 0 }
					byId[window.id] = row
					table.insert(rows, row)
				end
				row.requests += 1
				if record.result == 'hit' then row.hit = true row.damage += record.damage or 0
				elseif record.result == 'dead' then row.dead = true
				elseif record.result == 'pending' then row.pending = true end
			end
		end
		table.sort(rows, function(a, b) return a.id < b.id end)
		for _, row in ipairs(rows) do
			row.result = row.hit and 'hit' or row.pending and 'pending' or row.dead and 'dead' or 'miss'
		end
		return rows
	end

	local function shortText(text)
		local ok, stop = pcall(utf8.offset, text, 25)
		if not ok then return string.sub(text, 1, 24) end
		if stop then return string.sub(text, 1, stop - 1) .. '…' end
		return text
	end
	local function infoText(value)
		local kind = typeof(value)
		if kind == 'number' then return string.format('%.6g', value)
		elseif kind == 'boolean' or kind == 'EnumItem' then return tostring(value)
		elseif kind == 'string' then return '"' .. shortText(value) .. '"'
		elseif kind == 'Vector3' or kind == 'Vector2' or kind == 'NumberRange' or kind == 'Color3' then return '(' .. tostring(value) .. ')'
		elseif kind == 'Instance' then return '<' .. value.ClassName .. '>' end
		return nil
	end
	local function gunInfoText(gun)
		local saved = savedInfo[gun]
		local entries = {}
		local function add(path, value, changed)
			local ok, text = pcall(infoText, value)
			if not ok or text == nil then return end
			local lower, rank = string.lower(path), 1
			for _, word in ipairs(INFO_WORDS) do
				if string.find(lower, word, 1, true) then rank = 0 break end
			end
			table.insert(entries, { path = path, text = text .. (changed and '*' or ''), rank = rank })
		end
		pcall(function()
			for key, value in pairs(gun) do
				local path = tostring(key)
				local changed = saved ~= nil and saved[key] ~= nil
				if changed then value = saved[key] or nil end
				if type(value) == 'table' then
					local n = 0
					for innerKey, innerValue in pairs(value) do
						n += 1
						if n > 40 then break end
						if type(innerValue) ~= 'table' then add(path .. '.' .. tostring(innerKey), innerValue, false) end
					end
				elseif value ~= nil then add(path, value, changed) end
			end
		end)
		if #entries == 0 then return '(읽을 값 없음)' end
		table.sort(entries, function(a, b)
			if a.rank ~= b.rank then return a.rank < b.rank end
			return a.path < b.path
		end)
		local parts = {}
		for i = 1, math.min(#entries, INFO_KEYS) do table.insert(parts, entries[i].path .. '=' .. entries[i].text) end
		return table.concat(parts, ', ') .. (#entries > INFO_KEYS and string.format(' 외 %d개', #entries - INFO_KEYS) or '')
	end

		local kills, deaths = 0, 0
		for _, entry in ipairs(Events.list) do
			if entry.kind == '킬' then kills += 1
			elseif entry.kind == '죽음' then deaths += 1 end
		end
		if kills + deaths > 0 then
			table.insert(lines, string.format('사건: 킬 %d  죽음 %d', kills, deaths))
		end
		return table.concat(lines, '\n'), lines
	end

	function Stats:Save(text)
		if type(writefile) ~= 'function' then return false end
		return (pcall(function() writefile('ygj_match_' .. tostring(Events.id) .. '.txt', text) end))
	end

	function Stats:HookError(err)
		local message = tostring(err)
		local ok, stop = pcall(utf8.offset, message, 301)
		if not ok then message = string.sub(message, 1, 300)
		elseif stop then message = string.sub(message, 1, stop - 1) end
		self.hookErrors += 1
		for _, entry in ipairs(self.hookErrorList) do
			if entry.text == message then entry.n += 1 return end
		end
		if #self.hookErrorList >= 5 then return end
		table.insert(self.hookErrorList, { text = message, n = 1 })
		warn('[YGJ] 샷 훅 오류 (이 샷은 게임 원래대로): ' .. message)
		event('훅 오류', message)
	end

	function Stats:SendWebhook(text, fileName)
		local url = env.YGJWebhookUrl
		if type(url) ~= "string" or url == "" then return false end
		local req = request or http_request or (syn and syn.request) or (http and http.request)
		if type(req) ~= "function" then return false end
		local HttpService = game:GetService("HttpService")
		local boundary = "----YGJBoundary" .. tostring(os.time()) .. tostring(math.random(100000, 999999))
		local payload = HttpService:JSONEncode({
			content = string.format("[YGJ 체크포인트 %d] 판 종료: 샷 %d발 (%s)", SCRIPT_CHECKPOINT, #self.log, fileName),
		})
		local body = table.concat({
			"--" .. boundary,
			'Content-Disposition: form-data; name="payload_json"',
			"Content-Type: application/json",
			"",
			payload,
			"--" .. boundary,
			'Content-Disposition: form-data; name="files[0]"; filename="' .. fileName .. '"',
			"Content-Type: text/plain; charset=utf-8",
			"",
			text,
			"--" .. boundary .. "--",
			"",
		}, "\r\n")
		task.spawn(function()
			local ok = pcall(req, { Url = url, Method = "POST", Headers = { ["Content-Type"] = "multipart/form-data; boundary=" .. boundary }, Body = body })
			if not ok then warn("[YGJ] 웹훅 실패") end
		end)
		return true
	end

	function Stats:Dump()
		local text = self:Text()
		self:Save(text)
		self:SendWebhook(text, 'ygj_cp' .. SCRIPT_CHECKPOINT .. '_dump.txt')
		return text
	end

	function Stats:NewMatch()
		if #self.log > 0 or #Events.list > 0 then
			local text = self:Text()
			self:Save(text)
			if #self.log > 0 or self.hookErrors > 0 then
				self:SendWebhook(text, 'ygj_cp' .. SCRIPT_CHECKPOINT .. '_match_' .. tostring(Events.id) .. '.txt')
			end
		end
		self.log = {}
		self.nearArm = NEAR_EXPERIMENT and (math.random() < 0.5 and 'move' or 'stay') or nil
		self.projArm = PROJ_EXPERIMENT and PROJ_ARMS[math.random(#PROJ_ARMS)] or nil
		table.clear(self.lateWatch)
		Events.list = {}
		Events.start = env.RivalsClock()
		Events.id = os.time()
		event('판 시작', tostring(LocalPlayer:GetAttribute('SpectatingDuelObjectID')))
	end

	function Stats:Rate()
		local judged = self.windowHits + self.windowMisses
		if judged > 0 then return self.windowHits / judged end
		local done = self.hits + self.misses
		if done == 0 then return nil end
		return self.hits / done
	end

	local lastMatch = LocalPlayer:GetAttribute('SpectatingDuelObjectID')
	Stats.connection = RunService.Heartbeat:Connect(function()
		local match = LocalPlayer:GetAttribute('SpectatingDuelObjectID')
		if match ~= lastMatch then lastMatch = match Stats:NewMatch() end
		Stats:Update()
		if v65('log') then pcall(Stats.Sample, Stats) end
		Stats:CheckHook()
	end)
end

local LEAD_LIMIT = 12
local BALLISTIC_LEAD_LIMIT = 40

local function ballisticPoint(origin, point, velocity, speed, gravity, ping)
	local lag = (ping or 100) / 1000
	local t = (point - origin).Magnitude / speed
	local future = point
	for _ = 1, 4 do
		future = point + velocity * (t + lag)
		t = (future - origin).Magnitude / speed
	end
	local shift = future - point
	if shift.Magnitude > BALLISTIC_LEAD_LIMIT then shift = shift.Unit * BALLISTIC_LEAD_LIMIT end
	return point + shift + Vector3.new(0, 0.5 * gravity * t * t, 0)
end

local function aimPoint(arm, base, targetHead, targetRoot, ping)
	local velocity = targetRoot and targetRoot.AssemblyLinearVelocity
	if arm == 'center' or not velocity then return base, base end
	if arm == 'lead' then
		local size = targetHead.Size
		local limit = 0.45 * math.min(size.X, size.Y, size.Z)
		local center = targetHead.Position
		local inside = base - center + velocity * ((ping or 80) / 1000)
		if inside.Magnitude > limit then inside = inside.Unit * limit end
		return center + inside, center + inside
	end
	local shift = velocity * ((ping or 120) / 1000 + (arm == 'ahead' and 0.1 or 0))
	if shift.Magnitude > LEAD_LIMIT then shift = shift.Unit * LEAD_LIMIT end
	return base + shift, base
end

local CLEAR_OFFSETS = {
	Vector3.new(0, 5, 0), Vector3.new(5, 2, 0), Vector3.new(-5, 2, 0), Vector3.new(0, 2, 5), Vector3.new(0, 2, -5), Vector3.new(0, -5, 0),
}

local function clearOrigin(targetHead)
	local head = targetHead.Position
	local candidates = {}
	local camera = currentCamera()
	if camera then
		local toward = camera.CFrame.Position - head
		if toward.Magnitude > 0.1 then table.insert(candidates, head + toward.Unit * 5) end
	end
	for _, offset in ipairs(CLEAR_OFFSETS) do table.insert(candidates, head + offset) end
	for _, point in ipairs(candidates) do
		if DesyncController:Sight(point, head, targetHead) and DesyncController:Free(point, targetHead.Parent) then return point end
	end
	return nil
end

local FLANK_REACH = 4
local FLANK_REACHES = { near = FLANK_REACH, mid = 7, far = 10 }
local FLANK_STEPS = 12

local function flankOrigin(player, targetHead, weaponName, planned, reach)
	local head = targetHead.Position
	local radius = reach or FLANK_REACH
	local best, bestGap = nil, math.huge
	for step = 0, FLANK_STEPS - 1 do
		local angle = step * (2 * math.pi / FLANK_STEPS)
		for _, lift in ipairs({ 0, 2 }) do
			local point = head + Vector3.new(math.cos(angle) * radius, lift, math.sin(angle) * radius)
			local gap = (point - planned).Magnitude
			if gap < bestGap
				and not env.RivalsTargetState:Blocked(player, point, weaponName)
				and DesyncController:Sight(point, head, targetHead)
				and DesyncController:Free(point, targetHead.Parent) then
				best, bestGap = point, gap
			end
		end
	end
	return best
end

local function sightPoint(from, targetHead)
	for _, point in ipairs(headPoints(targetHead)) do
		if DesyncController:Sight(from, point, targetHead) then return point end
	end
	return targetHead.Position
end

local boxParams = RaycastParams.new()
boxParams.FilterType = Enum.RaycastFilterType.Exclude
boxParams.IgnoreWater = true
boxParams.RespectCanCollide = true

local function boxCentre(from, box, character)
	local filter = {}
	if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
	local camera = currentCamera()
	if camera then table.insert(filter, camera) end
	boxParams.FilterDescendantsInstances = filter
	local target = box.Position
	local result = Workspace:Raycast(from, target - from, boxParams)
	if result == nil or result.Instance:IsDescendantOf(character) then return target end
	return nil
end

local function findShot(result)
	local data = result[3]
	if typeof(data) == 'table' then return data, false end
	for i = 1, result.n do
		local value = result[i]
		if typeof(value) == 'table' and (value[utf8.char(0)] ~= nil or value[utf8.char(1)] ~= nil) then return value, true end
	end
	return nil, false
end

local function shotTable(result)
	local data, relocated = findShot(result)
	if relocated then env.RivalsStats.relocated += 1 end
	return data
end

local function shapeOf(result)
	local parts = {}
	for i = 1, math.min(result.n, 6) do table.insert(parts, typeof(result[i])) end
	return result.n .. '개' .. (#parts > 0 and ('(' .. table.concat(parts, ',') .. ')') or '')
end

local NEAR_DISTANCE = 4
local REACH_MIN = 50
local HOLD_FRAMES = 6
local WARM_UP = 0.3
local SLOW_CYCLE = 0.25
local FAST_WARM = 0.13
local PREWARM_MAX = 0.6
local PREWARM_DRY = 0.4
local LINGER_MIN, LINGER_MAX, LINGER_EXTRA = 0.35, 0.8, 0.2

local function antiAimMoved()
	local frameAt = env.RivalsAntiAimFrameAt
	return env.RivalsAntiAimMoved == true and frameAt ~= nil and env.RivalsClock() - frameAt < 0.1
end
local MELEE_ESCAPE = 18

local function shotOrigin(arm, targetHead, targetRoot)
	local camera = currentCamera()
	if arm == 'camera' and camera then return camera.CFrame.Position
	elseif arm == 'near' and camera then
		local toward = camera.CFrame.Position - targetHead.Position
		if toward.Magnitude > NEAR_DISTANCE then return targetHead.Position + toward.Unit * NEAR_DISTANCE end
		return camera.CFrame.Position
	elseif arm == 'clear' then
		local point = clearOrigin(targetHead)
		if point then return point end
	elseif arm == 'shooter' then
		local placed = DesyncController.active and DesyncController.placed
		if placed then return placed.Position + Vector3.new(0, 1.75, 0) end
		if camera then return camera.CFrame.Position end
	end
	return targetHead.Position - Vector3.new(0, 8, 0)
end

local ShotWindow = { count = 0 }
local SWITCH_DIMS = { hitpart = true, origin = true, point = true, direct = true }
env.RivalsShotWindow = ShotWindow
do
	local function realCycle(controller)
		local info = type(controller) == 'table' and rawget(controller, 'Info') or nil
		if info == nil then pcall(function() info = controller.Info end) end
		if type(info) ~= 'table' then return 0 end
		local original = savedInfo[info]
		local function value(key)
			local saved = original and original[key]
			local found = saved
			if saved == nil then found = info[key] end
			return type(found) == 'number' and found or 0
		end
		local cooldown = value('ShootCooldown')
		local burst = math.max(1, value('ShootBurst'))
		local burstCooldown = value('ShootBurstCooldown')
		return (burst - 1) * burstCooldown + math.max(cooldown, burstCooldown)
	end
	ShotWindow.RealCycle = realCycle

	function ShotWindow:Enter(controller, now)
		local current = self.current
		if current and current.controller == controller and now < current.endsAt then
			current.requests += 1
			return current
		end
		self.count += 1
		local weapon = nil
		pcall(function() weapon = controller.Name end)
		current = { id = self.count, controller = controller, weapon = type(weapon) == 'string' and weapon or nil, startsAt = now, endsAt = now + realCycle(controller), requests = 1, choices = {} }
		self.current = current
		return current
	end

	function ShotWindow:Choose(window, name)
		local choice = window.choices[name]
		if choice == nil then
			local weapon = window.weapon
			if weapon and window.ctx and CTX_DIMS[name] then weapon = weapon .. window.ctx end
			local mode
			choice, mode = env.RivalsTuner:Choose(name, weapon)
			local from = window.switchFrom
			if from and SWITCH_DIMS[name] and from[name] ~= nil and choice == from[name] then
				local other = env.RivalsTuner:BestOther(name, weapon, choice)
				if other then choice, mode = other, 'switch' end
			end
			window.choices[name] = choice
			window.modes = window.modes or {}
			window.modes[name] = mode
		end
		return choice
	end
end

env.WallbangController = {}
do
	function WallbangController:init()
		self.active = true
		self.startShootingRef = GunItem and GunItem.StartShooting
		self.desyncCleanup = nil
		self:Start()
	end

	function WallbangController:ArmCleanup(linger)
		linger = linger or 0.15
		DesyncController.expireAt = env.RivalsClock() + math.max(0.6, linger + 0.1)
		if self.desyncCleanup then pcall(task.cancel, self.desyncCleanup) end
		self.desyncCleanup = task.delay(linger, function() DesyncController:Stop('idle') end)
	end

	local function lingerFor(controller)
		local cycle = ShotWindow.RealCycle(controller)
		if cycle ~= cycle then cycle = 0 end
		return math.clamp(cycle + LINGER_EXTRA, LINGER_MIN, LINGER_MAX)
	end

	function WallbangController:MeleeSpots(target)
		if not env.RivalsTargetState:HoldsMelee(target) then return false, nil end
		local character = target.Character
		local root = character and character:FindFirstChild('HumanoidRootPart')
		if not root or not v65('meleeStill') or root.AssemblyLinearVelocity.Magnitude >= STILL_SPEED then return true, nil end
		local tuner = env.RivalsTuner
		if tuner.meleeStillCharacter ~= character then
			tuner.meleeStillCharacter = character
			tuner.meleeStillArm = tuner:Choose('meleeStill')
		end
		return tuner.meleeStillArm ~= 'near', tuner.meleeStillArm
	end

	function WallbangController:ArmFor(target)
		local tuner = env.RivalsTuner
		if tuner.desyncTarget ~= target or tuner.desyncCharacter ~= target.Character then
			tuner.desyncTarget = target
			tuner.desyncCharacter = target.Character
			tuner.desyncArm = tuner:Choose('desync')
		end
		return tuner.desyncArm
	end

	function WallbangController:Start()
		self:Stop()
		if type(self.startShootingRef) ~= 'function' or not Utility or type(Utility.EncodeCFrame) ~= 'function' then
			self.active = false
			return
		end
		self.active = true
		GunItem.StartShooting = function(controller, ...)
			local result = table.pack(self.startShootingRef(controller, ...))
			local data = findShot(result)
			local before = nil
			if type(data) == 'table' then
				before = {}
				for key, value in next, data do before[key] = value end
			end
			local spread = result[4]
			local ok, err = pcall(self.Aim, self, controller, result)
			if not ok then
				pcall(function()
					if before then
						for key in next, data do data[key] = nil end
						for key, value in next, before do data[key] = value end
					end
					result[4] = spread
				end)
				pcall(env.RivalsStats.HookError, env.RivalsStats, err)
			end
			return unpack(result, 1, result.n)
		end
	end

	function WallbangController:Aim(controller, result)
		local clientFighter = controller.ClientFighter
		if not clientFighter.IsLocalPlayer then return end
		env.RivalsStats.hookCalls += 1

		local cameraData = shotTable(result)
		if not cameraData then
			env.RivalsStats:Skip('payload', shapeOf(result))
			return
		end
		env.RivalsShotAt = env.RivalsClock()

		local hookWait = nil
		if antiAimMoved() then
			env.RivalsStats.heldShots += 1
			local began = env.RivalsClock()
			for _ = 1, HOLD_FRAMES do
				self.holdUntil = math.max(self.holdUntil or 0, env.RivalsClock() + 0.05)
				task.wait()
				if not antiAimMoved() then break end
			end
			hookWait = env.RivalsClock() - began
		end

		local window = ShotWindow:Enter(controller, env.RivalsClock())
		local function choose(name) return ShotWindow:Choose(window, name) end

		if on('noSpread') then result[4] = true end

		local rage = on('rage')
		local silent = not rage and on('silent')
		local targetPlayer = TargetController.target
		if silent and targetPlayer == nil then targetPlayer = TargetController:VisibleEnemy() end
		if not self.active or not (rage or silent) or not targetPlayer or not targetPlayer.Character then
			env.RivalsStats:Skip(not (rage or silent) and 'rageOff' or 'noTarget')
			return
		end

		if self.desyncCleanup then task.cancel(self.desyncCleanup) self.desyncCleanup = nil end
		DesyncController.expireAt = env.RivalsClock() + 0.6

		local weaponName = nil
		pcall(function() weaponName = controller.Name end)
		local projectile = false
		pcall(function() projectile = controller.Info.IsProjectile == true end)
		local slowGun = ShotWindow.RealCycle(controller) >= SLOW_CYCLE
		local visible, inRange, distance, seenPoint = TargetController:Judge(targetPlayer)
		local pinned = false
		if silent and not visible then
			local other, otherPoint, otherDistance = TargetController:VisibleEnemy()
			if not other then
				local bodyPlayer, bodyPart, bodyPoint = TargetController:VisibleBodyEnemy(weaponName)
				if bodyPlayer then self:BodyShot(controller, cameraData, bodyPlayer, bodyPart, bodyPoint, weaponName, window) return end
				env.RivalsStats:Skip('noTarget')
				return
			end
			targetPlayer, seenPoint, distance = other, otherPoint, otherDistance
			visible, inRange, pinned = true, true, true
		elseif silent and targetPlayer ~= TargetController.target then
			pinned = true
		end
		if not visible and not inRange then
			env.RivalsStats:Skip('noTarget')
			self:ArmCleanup()
			return
		end
		local direct = visible
		local reachArm = nil
		local reachDim = nil
		local escape = rage and visible and distance < MELEE_ESCAPE and env.RivalsTargetState:HoldsMelee(targetPlayer)
		if rage and visible and inRange and not escape and slowGun then
			reachDim = distance >= REACH_MIN and 'reach' or 'reachNear'
			reachArm = (reachDim == 'reachNear' and env.RivalsStats.nearArm) or choose(reachDim)
			if reachArm ~= 'move' and reachArm ~= 'stay' then reachArm = nil end
		end
		local shieldSide = nil
		if rage and v65('shieldMove') then
			local camera = currentCamera()
			shieldSide = camera and env.RivalsTargetState:Blocked(targetPlayer, camera.CFrame.Position, weaponName) or nil
		end
		local warmReuse = false
		local warmSince = DesyncController.placedAt
		if rage and visible and not escape and v65('warmReuse') and DesyncController.active and DesyncController.startTarget == targetPlayer
			and DesyncController.placed ~= nil and warmSince ~= nil and env.RivalsClock() - warmSince >= WARM_UP then
			warmReuse = true
		end
		local desyncArm = (visible and reachArm ~= 'move' and not escape and not shieldSide and not warmReuse) and 'none' or self:ArmFor(targetPlayer)
		DesyncController.avoidBlock = (rage and v65('shieldMove')) and targetPlayer or nil
		DesyncController.blockWeapon = weaponName
		local meleeSafe, meleeStillArm = false, nil
		local warmed = nil
		if desyncArm == 'none' then
			if DesyncController.active then DesyncController:Stop('visible') end
		else
			meleeSafe, meleeStillArm = self:MeleeSpots(targetPlayer)
			if escape then meleeSafe, meleeStillArm = true, nil end
			local spots = DESYNC_CANDIDATES[meleeSafe and 'melee' or desyncArm]
			if DesyncController.offsets ~= spots then
				DesyncController.pick = nil
				if DesyncController.placedAt then DesyncController.placedAt = env.RivalsClock() end
			end
			DesyncController.offsets = spots
			DesyncController.fallback = DESYNC_FALLBACK[meleeSafe and 'melee' or desyncArm]
			if not DesyncController.active or DesyncController.startTarget ~= targetPlayer then
				DesyncController.offset = spots[1]
				DesyncController:Start(targetPlayer)
			end
			local ready = (DesyncController.startedAt or 0) + 0.1 - env.RivalsClock()
			if ready > 0 then
				self.holdUntil = math.max(self.holdUntil or 0, env.RivalsClock() + ready + 0.02)
				task.wait(ready)
				hookWait = (hookWait or 0) + ready
			end
			local warmFor = WARM_UP
			if slowGun and not escape and v65('hookWarm') then
				local began = env.RivalsClock()
				local waited = nil
				while DesyncController.active and env.RivalsClock() - began < warmFor do
					local placedAt = DesyncController.placedAt
					if placedAt and env.RivalsClock() - placedAt >= warmFor then break end
					DesyncController.expireAt = math.max(DesyncController.expireAt or 0, env.RivalsClock() + 0.6)
					self.holdUntil = math.max(self.holdUntil or 0, env.RivalsClock() + 0.05)
					task.wait()
					waited = env.RivalsClock() - began
				end
				if waited then
					warmed = waited
					hookWait = (hookWait or 0) + waited
					env.RivalsStats.warms += 1
					env.RivalsStats.warmWait += waited
				end
			end
		end
		if v65('placeFallback') and desyncArm ~= 'none' and DesyncController.placed == nil and visible and not escape and not (reachDim == 'reachNear' and env.RivalsStats.nearArm) then
			desyncArm, reachArm, meleeSafe = 'none', 'stay', false
			DesyncController.offsets, DesyncController.fallback = {}, {}
		end

		local character = targetPlayer.Character
		local stillTarget = pinned or TargetController.target == targetPlayer
		local retargeted = false
		if not self.active then
			env.RivalsStats:Skip('noTarget')
			self:ArmCleanup()
			return
		end
		if not stillTarget or not character or not character:FindFirstChild('Head') or not TargetController:IsShootable(targetPlayer, true) then
			local other = TargetController.target
			local otherCharacter = other and other.Character
			if other and otherCharacter and otherCharacter:FindFirstChild('Head') and TargetController:IsShootable(other, true) then
				targetPlayer, character, retargeted = other, otherCharacter, true
				local _, _, _, otherPoint = TargetController:Judge(other)
				seenPoint = otherPoint
				direct = true
				reachArm = nil
				escape = false
				desyncArm = 'none'
				shieldSide = nil
				DesyncController.avoidBlock = nil
				if DesyncController.active then DesyncController:Stop('retarget') end
			else
				env.RivalsStats:Skip('noTarget')
				self:ArmCleanup()
				return
			end
		end

		env.RivalsStats:CheckPayload(cameraData)
		local targetHead = character.Head
		local targetRoot = character:FindFirstChild('HumanoidRootPart')
		if window.ctx == nil and v65('ctx') and targetRoot then
			if v65('crouchCtx') and env.RivalsTargetState:Crouching(targetPlayer) then window.ctx = '#crouch'
			else window.ctx = targetRoot.AssemblyLinearVelocity.Magnitude < STILL_SPEED and '#still' or '' end
		end
		if window.switchFrom == nil then
			window.switchFrom = false
			local from = V65.switch == true and env.RivalsStats:MissRun(targetPlayer) or nil
			if from then
				window.switchTried = true
				if env.RivalsTuner.random() < 0.5 then window.switchFrom = from end
			end
		end
		local pointArm = choose('point')

		local shieldMove = shieldSide ~= nil and desyncArm ~= 'none'
		warmReuse = warmReuse and desyncArm ~= 'none'
		local meleeShooter = meleeSafe and desyncArm ~= 'none' and v65('meleeShooter')
		local forcedShooter = shieldMove or warmReuse or meleeShooter
		local originArm
		if retargeted then originArm = 'camera'
		elseif forcedShooter or (direct and (reachArm == 'move' or escape)) then originArm = 'shooter'
		elseif direct then originArm = choose('direct')
		else originArm = choose('origin') end
		local shootingPos = shotOrigin(originArm, targetHead, targetRoot)
		local targetState = env.RivalsTargetState
		local bodyAt = shootingPos
		if v65('shieldMove') then
			local placed = desyncArm ~= 'none' and DesyncController.active and DesyncController.placed or nil
			local camera = currentCamera()
			if placed then bodyAt = placed.Position + Vector3.new(0, 1.75, 0)
			elseif desyncArm == 'none' and camera then bodyAt = camera.CFrame.Position end
		end
		local block = targetState:Blocked(targetPlayer, bodyAt, weaponName)
		local flank, control, flankArm = nil, nil, nil
		if block then
			targetState.blocked += 1
			if block ~= 'deflect' and env.RivalsTuner.random() < targetState.BLOCK_CONTROL then control = true
			else
				local reachName = v65('flankReach') and choose('flankReach') or nil
				local alternative = flankOrigin(targetPlayer, targetHead, weaponName, shootingPos, reachName and FLANK_REACHES[reachName] or nil)
				if alternative then
					shootingPos = alternative
					flank = true
					flankArm = reachName
				end
			end
		end
		local fromCamera = direct and reachArm ~= 'move' and not escape and not forcedShooter
		local basePos = (fromCamera and not flank) and seenPoint or sightPoint(shootingPos, targetHead)
		local partArm = choose('hitpart')
		local headBox = nil
		if partArm == 'hitbox' or partArm == 'box' then
			local box = character:FindFirstChild(weaponName == 'Sniper' and 'HitboxHeadSmall' or 'HitboxHead')
			if box and box:IsA('BasePart') then headBox = box else partArm = nil end
		elseif partArm ~= 'head' then partArm = nil end
		local boxed = nil
		if partArm == 'box' and headBox and not projectile then
			boxed = boxCentre(shootingPos, headBox, character)
			if boxed then basePos = boxed end
		end
		local edge = not boxed and (basePos - targetHead.Position).Magnitude > 0.01
		local hitPos, markPos = aimPoint(pointArm, basePos, targetHead, targetRoot, env.RivalsStats:Ping())
		if window.scanOffset == nil then
			local random = env.RivalsTuner.random
			if window.ctx == '#still' and v65('scan') and random() < 0.3 then
				window.scanOffset = Vector3.new((random() - 0.5) * 0.5, (random() - 0.5) * 0.5, (random() - 0.5) * 0.5)
			else window.scanOffset = false end
		end
		local scanApplied = (window.scanOffset and not edge) and window.scanOffset or nil
		if scanApplied then hitPos, markPos = hitPos + scanApplied, markPos + scanApplied end
		local ballistic = nil
		if v65('ballistic') and projectile and targetRoot and (weaponName == 'RPG' or env.RivalsStats.projArm == 'real') then
			pcall(function()
				local gun = controller.Info
				local saved = savedInfo[gun]
				local speed = saved and type(saved.ProjectileSpeed) == 'number' and saved.ProjectileSpeed or gun.ProjectileSpeed
				if type(speed) == 'number' and speed > 1 and speed < 5000 then
					local gravity = (tonumber(gun.ProjectileGravity) or 0) * Workspace.Gravity
					hitPos = ballisticPoint(shootingPos, hitPos, targetRoot.AssemblyLinearVelocity, speed, gravity, env.RivalsStats:Ping())
					ballistic = true
				end
			end)
		end
		local hitPart = headBox or targetHead
		local shootingOffset = hitPart.CFrame:ToObjectSpace(CFrame.new(markPos))

		local facingArm = choose('facing')
		if projectile and v65('projectileLook') then facingArm = 'look' end
		local aimOffset = hitPos - shootingPos
		if facingArm == 'look' and aimOffset.Magnitude > 0.01 then
			cameraData[utf8.char(0)] = Utility:EncodeCFrame(CFrame.lookAt(shootingPos, hitPos))
			cameraData[utf8.char(1)] = Utility:EncodeCFrame(CFrame.lookAt(hitPos, hitPos + aimOffset))
		else
			cameraData[utf8.char(0)] = Utility:EncodeCFrame(CFrame.new(shootingPos, hitPos) * CFrame.Angles(CFrame.lookAt(shootingPos, hitPos):ToOrientation()))
			cameraData[utf8.char(1)] = Utility:EncodeCFrame(CFrame.new(hitPos) * CFrame.Angles(CFrame.lookAt(shootingPos, hitPos):ToOrientation()))
		end
		cameraData[utf8.char(2)] = hitPart
		cameraData[utf8.char(3)] = Utility:EncodeCFrame(shootingOffset)

		local moved, exact, approached = nil, nil, nil
		if desyncArm ~= 'none' then
			moved = DesyncController.active == true and DesyncController.placed ~= nil
			exact = (moved and DesyncController.placedExact == true) or nil
			local pick = moved and DesyncController.pick or nil
			approached = (pick and pick.list and pick.list.anchor == 'flat') or nil
		end
		local arms = { point = pointArm, facing = facingArm, hitpart = partArm }
		if reachDim then arms[reachDim] = reachArm end
		if not direct then
			arms.origin = originArm
			arms.desync = desyncArm
		elseif not retargeted and reachArm ~= 'move' and not escape then
			arms.direct = originArm
		end
		if flank then
			arms.origin = nil
			arms.direct = nil
			arms.flankReach = flankArm
		end
		if forcedShooter then
			arms.origin, arms.direct = nil, nil
			if shieldMove or warmReuse then
				arms.desync = nil
				if reachDim then arms[reachDim] = nil end
			end
		end
		if meleeStillArm and desyncArm ~= 'none' then arms.meleeStill = meleeStillArm end
		if meleeSafe then arms.desync = nil end
		local prewarmHeld = nil
		if self.prewarmDoneAt and env.RivalsClock() - self.prewarmDoneAt < 0.25 then
			prewarmHeld = self.prewarmHeld
			self.prewarmDoneAt = nil
		end
		local prewarmed = (desyncArm ~= 'none' and DesyncController.active and DesyncController.prewarmed) or nil
		local extra = nil
		if v65('log') then
			extra = { pw = prewarmHeld, dh = hookWait, wu = warmed, fr = flankArm, pa = projectile and env.RivalsStats.projArm or nil, bl = ballistic and '1' or nil }
			if scanApplied then extra.ox, extra.oy, extra.oz = scanApplied.X, scanApplied.Y, scanApplied.Z end
			local placedAt = DesyncController.placedAt
			if desyncArm ~= 'none' and DesyncController.active and placedAt then extra.da = env.RivalsClock() - placedAt end
			pcall(function()
				local box = character:FindFirstChild('HitboxHead')
				if box and box:IsA('BasePart') then
					extra.hb = (box.Position - targetHead.Position).Magnitude
					extra.hy = box.Position.Y - targetHead.Position.Y
				end
			end)
			pcall(function()
				local placed = DesyncController.placed
				local ownRoot = Character.RootPart
				extra.dO = targetState:LookDot(targetPlayer, shootingPos)
				extra.dP = placed and targetState:LookDot(targetPlayer, placed.Position) or nil
				extra.od = ownRoot and (shootingPos - ownRoot.Position).Magnitude or nil
				extra.pd = placed and (shootingPos - placed.Position).Magnitude or nil
				extra.ms = ownRoot and ownRoot.AssemblyLinearVelocity.Magnitude or nil
				extra.cr = targetState:Crouching(targetPlayer)
			end)
		end
		local weapon = type(controller) == 'table' and controller.Info or controller
		env.RivalsStats:Record(targetPlayer, { arms = arms, moved = moved, exact = exact, direct = direct or nil, visible = visible or nil, retargeted = retargeted or nil, edge = edge or nil, weapon = weapon, gun = type(weapon) == 'table' and weapon ~= controller and weapon or nil, weaponName = weaponName ~= nil and tostring(weaponName) or nil, silent = silent or nil, meleeSafe = meleeSafe or nil, escape = escape or nil, invincible = targetState:IsInvincible(targetPlayer) or nil, block = block, flank = flank, control = control, shield = shieldMove and shieldSide or nil, warm = warmReuse or nil, approach = approached or nil, prewarm = prewarmed, window = window, x = extra, scanned = scanApplied and true or nil })
		self:ArmCleanup((desyncArm ~= 'none' and v65('linger')) and lingerFor(controller) or nil)
	end

	function WallbangController:BodyShot(controller, cameraData, targetPlayer, part, point, weaponName, window)
		local camera = currentCamera()
		local origin = camera and camera.CFrame.Position or point
		local aimOffset = point - origin
		if aimOffset.Magnitude > 0.01 then
			if env.RivalsTuner:Best('facing') == 'look' then
				cameraData[utf8.char(0)] = Utility:EncodeCFrame(CFrame.lookAt(origin, point))
				cameraData[utf8.char(1)] = Utility:EncodeCFrame(CFrame.lookAt(point, point + aimOffset))
			else
				cameraData[utf8.char(0)] = Utility:EncodeCFrame(CFrame.new(origin, point) * CFrame.Angles(CFrame.lookAt(origin, point):ToOrientation()))
				cameraData[utf8.char(1)] = Utility:EncodeCFrame(CFrame.new(point) * CFrame.Angles(CFrame.lookAt(origin, point):ToOrientation()))
			end
		end
		cameraData[utf8.char(2)] = part
		cameraData[utf8.char(3)] = Utility:EncodeCFrame(part.CFrame:ToObjectSpace(CFrame.new(point)))
		local weapon = type(controller) == 'table' and controller.Info or controller
		env.RivalsStats:Record(targetPlayer, { direct = true, visible = true, silent = true, body = true, weapon = weapon, gun = type(weapon) == 'table' and weapon ~= controller and weapon or nil, weaponName = weaponName ~= nil and tostring(weaponName) or nil, window = window })
	end

	function WallbangController:Prewarm()
		if not self.active then return false end
		local now = env.RivalsClock()
		local lastCall = self.prewarmCall
		self.prewarmCall = now
		if lastCall == nil or now - lastCall > 0.25 then self.pressFrom, self.prewarmFrom = nil, nil
		elseif self.pressFrom and (env.RivalsShotAt or -math.huge) >= self.pressFrom then self.pressFrom = nil end
		if now < (self.holdUntil or 0) then return true end
		local function press()
			local from = self.prewarmFrom
			if from then
				self.prewarmFrom = nil
				self.prewarmHeld, self.prewarmDoneAt = now - from, now
				env.RivalsStats.warms += 1
				env.RivalsStats.warmWait += now - from
			end
			self.pressFrom = self.pressFrom or now
			return false
		end
		if not on('rage') or not v65('prewarm') then return press() end
		local target = TargetController.target
		local character = target and target.Character
		if not character or not character:FindFirstChild('HumanoidRootPart') or TargetController.targetVisible ~= false or TargetController.targetInRange ~= true then return press() end
		local item = env.RivalsTargetState:Equipped(LocalPlayer)
		local isGun, weaponName = false, nil
		if item then
			pcall(function()
				isGun = type(item.StartShooting) == 'function'
				weaponName = item.Name
			end)
		end
		if not isGun then return press() end
		local cycle = ShotWindow.RealCycle(item)
		if cycle ~= cycle then cycle = 0 end
		local warmFor = cycle >= SLOW_CYCLE and WARM_UP or FAST_WARM
		if self.pressFrom and now - self.pressFrom > cycle + PREWARM_DRY then return press() end
		local desync = DesyncController
		if not desync.active or desync.startTarget ~= target then
			local list = self:MeleeSpots(target) and 'melee' or self:ArmFor(target)
			local spots = DESYNC_CANDIDATES[list]
			if not spots then return press() end
			desync.offsets, desync.fallback, desync.offset = spots, DESYNC_FALLBACK[list], spots[1]
			desync.avoidBlock = v65('shieldMove') and target or nil
			desync.blockWeapon = weaponName
			desync:Start(target)
			desync.prewarmed = true
		end
		desync.expireAt = math.max(desync.expireAt or 0, now + 0.6)
		if self.desyncCleanup then pcall(task.cancel, self.desyncCleanup) self.desyncCleanup = nil end
		local placedAt = desync.placedAt
		if now - (desync.startedAt or now) >= PREWARM_MAX or (placedAt and now - placedAt >= warmFor) then return press() end
		self.prewarmFrom = self.prewarmFrom or now
		self.pressFrom = nil
		return true
	end

	function WallbangController:Stop()
		self.active = false
		if GunItem and self.startShootingRef then GunItem.StartShooting = self.startShootingRef end
		if self.desyncCleanup then pcall(task.cancel, self.desyncCleanup) self.desyncCleanup = nil end
	end

	WallbangController:init()
	env.RivalsPrewarm = function() return WallbangController:Prewarm() end
end

	background.BorderColor3 = Color3.fromRGB(0, 0, 0)
	background.Parent = billboard

	fill.BackgroundColor3 = BAR_FULL_COLOR
	fill.BorderSizePixel = 0
	fill.Parent = background

		fill.Size = UDim2.new(1, 0, ratio, 0)
		fill.BackgroundColor3 = BAR_EMPTY_COLOR:Lerp(BAR_FULL_COLOR, ratio)

		tracked[targetPlayer] = { highlight = highlight, bar = bar, hrp = hrp, humanoid = humanoid, char = char }
	end

	if targetPlayer == player then return end
	if targetPlayer.Character then task.spawn(hookCharacter, targetPlayer, targetPlayer.Character) end

	local filter = { camera }
	local myChar = player.Character
	if myChar then table.insert(filter, myChar) end
	for _, other in ipairs(Players:GetPlayers()) do
		local otherChar = other.Character
		if otherChar and otherChar ~= myChar then table.insert(filter, otherChar) end
	end
	return filter
end

	return workspace:Raycast(origin, part.Position - origin, rayParams) == nil

			local visible = canSee(origin, data.char:FindFirstChild("Head")) or canSee(origin, data.hrp)

		end
	end

local meleeCache = { at = -1, value = false }

local function probeMelee(item, info)
	if meleeSeen[info] then return end
	meleeSeen[info] = true
	pcall(function()
		local keys, parts = {}, {}
		for key in pairs(info) do table.insert(keys, tostring(key)) end
		table.sort(keys)
		for _, key in ipairs(keys) do
			local value = info[key]
			local kind = typeof(value)
			if kind == "number" or kind == "boolean" or kind == "string" then table.insert(parts, key .. "=" .. tostring(value)) end
		end
		local fns = {}
		local function collect(tbl)
			for key, value in pairs(tbl) do if type(value) == "function" then table.insert(fns, tostring(key)) end end
		end
		pcall(collect, item)
		local meta = getmetatable(item)
		if type(meta) == "table" and type(meta.__index) == "table" then pcall(collect, meta.__index) end
		table.sort(fns)
		local text = tostring(info.Name or rawget(item, "Name") or "melee") .. " | " .. table.concat(parts, ", ") .. " | 함수: " .. table.concat(fns, ", ")
		if type(writefile) == "function" then pcall(writefile, "ygj_melee_probe.txt", text) end
		if env.RivalsEvent then env.RivalsEvent("근접 정보", string.sub(text, 1, 3000)) end
	end)
end

local function meleeState()
	local now = env.RivalsClock()
	if now - meleeCache.at < 0.1 then return meleeCache.value or nil end
	meleeCache.at = now
	meleeCache.value = false
	if cfg.melee == false then return nil end
	local state = env.RivalsTargetState
	local item = state and state.Equipped and state:Equipped(LocalPlayer)
	if not item then return nil end
	local info = rawget(item, "Info")
	if info == nil then pcall(function() info = item.Info end) end
	if type(info) ~= "table" or info.Type ~= "Melee" then return nil end
	probeMelee(item, info)
	local reach = MELEE_REACH
	for _, key in ipairs(MELEE_REACH_KEYS) do
		local value = info[key]
		if type(value) == "number" and value > 1 and value < 60 then reach = value break end
	end
	meleeCache.value = { item = item, info = info, reach = reach }
	return meleeCache.value
end

local function meleeEnemy(melee, extra)
	local controller = env.TargetController
	if not controller or controller.active == false then return nil end
	local enemy = controller:NearestEnemy(melee.reach + extra)
	if not enemy or not controller:IsShootable(enemy, true) then return nil end
	if not controller:Judge(enemy) then return nil end
	return enemy
end

local function meleeWantsFire(melee)
	return meleeEnemy(melee, 0) ~= nil
end

local function rageWantsFire()
	if not cfg.autoFire or not (cfg.rage or cfg.silent) then return false end
	local wallbang = env.WallbangController
	if not (wallbang and wallbang.active) then return false end
	local controller = env.TargetController
	if controller == nil or controller.active == false or controller.target == nil then return false end
	if cfg.rage then return controller.targetVisible ~= false or controller.targetInRange ~= false end
	if controller.targetVisible == true then return true end
	return controller.VisibleBodyEnemy ~= nil and controller:VisibleBodyEnemy() ~= nil
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude
rayParams.IgnoreWater = true

local TRIGGER_SPREAD = 0.006
local TRIGGER_HOLD = 0.08
local TRIGGER_POINTS = {
	{ 0, 0 }, { 1, 0 }, { -1, 0 }, { 0, 1 }, { 0, -1 },
	{ 0.7, 0.7 }, { -0.7, 0.7 }, { 0.7, -0.7 }, { -0.7, -0.7 },
}
local triggerUntil = 0
env.RivalsTrigger = { hits = 0, held = 0, spread = 0, blocked = 0 }

local function enemyUnder(camera, x, y)
	local ray = camera:ViewportPointToRay(x, y)
	local hit = Workspace:Raycast(ray.Origin, ray.Direction * 2000, rayParams)
	if not hit then return false end
	local owner = ownerOf(hit.Instance)
	if owner == nil or not isEnemy(owner) or not isAlive(owner.Character) then return false end
	local targetState = env.RivalsTargetState
	if targetState and targetState:Unshootable(owner) then return false end
	if not (cfg.rage or cfg.silent) and targetState and targetState.Blocked and targetState:Blocked(owner, ray.Origin) then
		local stats = env.RivalsTrigger
		if stats then stats.blocked += 1 end
		return false
	end
	return true
end

local function triggerWantsFire()
	if not cfg.triggerbot then return false end
	local camera = Workspace.CurrentCamera
	if not camera then return false end
	local filter = { camera }
	if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
	for _, p in ipairs(Players:GetPlayers()) do
		local character = p ~= LocalPlayer and p.Character
		if character then
			for _, child in ipairs(character:GetChildren()) do
				if child:IsA("Accessory") then table.insert(filter, child) end
			end
		end
	end
	rayParams.FilterDescendantsInstances = filter
	local mouse = UserInputService:GetMouseLocation()
	local height = 1080
	pcall(function() height = camera.ViewportSize.Y end)
	local radius = math.max(3, (tonumber(height) or 1080) * TRIGGER_SPREAD)
	local stats = env.RivalsTrigger
	for index, point in ipairs(TRIGGER_POINTS) do
		if enemyUnder(camera, mouse.X + point[1] * radius, mouse.Y + point[2] * radius) then
			triggerUntil = env.RivalsClock() + TRIGGER_HOLD
			if stats then
				stats.hits += 1
				if index > 1 then stats.spread += 1 end
			end
			return true
		end
	end
	if env.RivalsClock() < triggerUntil then
		if stats then stats.held += 1 end
		return true
	end
	return false
end

local pauseUntil = 0
local function pauseFor(seconds)
	local untilAt = env.RivalsClock() + seconds
	if untilAt > pauseUntil then pauseUntil = untilAt end
end

local switchKeys = {}
for _, name in ipairs({
	"One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", "Zero",
	"Q", "E", "F", "G", "Tab",
	"DPadUp", "DPadDown", "DPadLeft", "DPadRight", "ButtonY",
}) do
	local ok, key = pcall(function() return Enum.KeyCode[name] end)
	if ok and key then switchKeys[key] = true end
end

end

local focusFn
if type(isrbxactive) == "function" then focusFn = isrbxactive
elseif type(iswindowactive) == "function" then focusFn = iswindowactive end

local focusFlag = true
table.insert(conns, UserInputService.WindowFocused:Connect(function() focusFlag = true end))
table.insert(conns, UserInputService.WindowFocusReleased:Connect(function()
	focusFlag = false
	setHeld(false)
end))

local function isFocused()
	if focusFn then
		local ok, value = pcall(focusFn)
		if ok then return value ~= false end
	end
	return focusFlag
end

table.insert(conns, LocalPlayer.CharacterAdded:Connect(function() pauseFor(SPAWN_GRACE) end))

local blockedTime = 0
local hinted = false
local wasCaptured = false
local wasFocused = true
local wasMenu = false
local lastTick = env.RivalsClock()

local seenShot = env.RivalsLastShotAt
local hookAlive = false
local trainStart = nil
local backoffUntil = 0

local function tick()
	cfg = env.RivalsCfg or cfg
	local now = env.RivalsClock()
	local dt = math.min(now - lastTick, 0.1)
	lastTick = now

	local focused = isFocused()
	if focused and not wasFocused then pauseFor(FOCUS_GRACE) end
	wasFocused = focused

	local captured = UserInputService.MouseBehavior ~= Enum.MouseBehavior.Default
	if captured and not wasCaptured then pauseFor(CAPTURE_GRACE) end
	wasCaptured = captured
	if captured then blockedTime = 0 end

	local menuOpen = false
	pcall(function() menuOpen = GuiService.MenuIsOpen == true end)

	local anyMenu = menuOpen or cfg.menuOpen == true
	if wasMenu and not anyMenu then pauseFor(FOCUS_GRACE) end
	wasMenu = anyMenu

	local usable = focused
		and not menuOpen
		and not cfg.menuOpen
		and UserInputService:GetFocusedTextBox() == nil
		and isAlive(LocalPlayer.Character)
		and now >= pauseUntil
		and now >= backoffUntil

	if not usable then
		setHeld(false)
		trainStart = nil
		return
	end

	local melee = meleeState()
	local wants
	if melee then wants = meleeWantsFire(melee) else wants = rageWantsFire() or triggerWantsFire() end
	if not wants then
		setHeld(false)
		trainStart = nil
		return
	end

	if not captured then
		blockedTime += dt
		if blockedTime >= HINT_AFTER and not hinted then
			hinted = true
			warn("[YGJ] 마우스가 안 잠겨 있어서 자동 발사를 안 한다. 경기 중인데도 이 말이 나오면 알려줘")
		end
		setHeld(false)
		trainStart = nil
		return
	end

	if not melee and env.RivalsPrewarm then
		local ok, wait = pcall(env.RivalsPrewarm)
		if ok and wait then
			setHeld(false)
			trainStart = nil
			return
		end
	end

	local lastShot = env.RivalsLastShotAt
	if lastShot ~= seenShot then
		seenShot = lastShot
		hookAlive = true
		trainStart = nil
	end
	if hookAlive and not melee then
		if not trainStart then trainStart = now
		elseif now - trainStart > NOSHOT_AFTER then
			backoffUntil = now + BACKOFF
			trainStart = nil
			setHeld(false)
			return
		end
	end

	setHeld(not held)
end

table.insert(conns, RunService.Heartbeat:Connect(function()
	if unloaded then return end
	local ok = pcall(tick)
	if not ok then setHeld(false) end
end))

local function meleeAim()
	if type(mousemoverel) ~= "function" then return end
	if not isFocused() or cfg.menuOpen or UserInputService:GetFocusedTextBox() ~= nil then return end
	if UserInputService.MouseBehavior == Enum.MouseBehavior.Default or env.RivalsClock() < pauseUntil then return end
	if not isAlive(LocalPlayer.Character) then return end
	local melee = meleeState()
	if not melee then return end
	local enemy = meleeEnemy(melee, MELEE_AIM_EXTRA)
	local root = enemy and enemy.Character and enemy.Character:FindFirstChild("HumanoidRootPart")
	local camera = Workspace.CurrentCamera
	if not (root and camera) then return end
	local view = camera:WorldToViewportPoint(root.Position + Vector3.new(0, 0.6, 0))
	if view.Z <= 0 then return end
	local size = camera.ViewportSize
	local dx, dy = view.X - size.X / 2, view.Y - size.Y / 2
	local length = math.sqrt(dx * dx + dy * dy)
	if length < 4 then return end
	local step = math.min(length * MELEE_AIM_GAIN, MELEE_AIM_MAX)
	mousemoverel(dx / length * step, dy / length * step)
end

table.insert(conns, RunService.RenderStepped:Connect(function()
	if unloaded then return end
	pcall(meleeAim)
end))

local function unload()
	if unloaded then return end
	unloaded = true
	for _, conn in ipairs(conns) do pcall(function() conn:Disconnect() end) end
	table.clear(conns)
	setHeld(false)
	if env.RivalsAutoFireUnload == unload then env.RivalsAutoFireUnload = nil end
end

env.RivalsAutoFireUnload = unload

end)
run("antiaim", function()

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

local env = getgenv()
local cfg = env.RivalsCfg or {}
local player = Players.LocalPlayer

local LEVELS = {
	{ spin = math.rad(1440), side = 4, upMin = 1, upMax = 4, farChance = 0, farMin = 0, farMax = 0 },
	{ spin = math.rad(2880), side = 6, upMin = 2, upMax = 8, farChance = 0.3, farMin = 8, farMax = 18 },
	{ spin = math.rad(6840), side = 10, upMin = 2, upMax = 12, farChance = 0.7, farMin = 15, farMax = 32 },
	{ spin = math.rad(6840), side = 10, upMin = 2, upMax = 12, farChance = 0.7, farMin = 15, farMax = 32, sink = env.RivalsKeepSink == true, tumble = env.RivalsKeepSink == true },
}
local JITTER_RANGE = 80
local JITTER_GROUND = 24
local RESTORE_NAME = "RivalsAntiAimRestore"
local TELEPORT_GAP = 70
local PULL_GAP = 4
local FLING_SPEED = 200
local PAUSE_AFTER = 0.5
local SHOT_HOLD = 0.4
local MELEE_NEAR = 18
local MELEE_SAFE = 14
local SINK_RANGE = 300
local SINK_MIN = 4
local SINK_MAX = 6
local KILL_RADIUS = 3
local DODGE_TURNS = { 0, math.rad(40), -math.rad(40), math.rad(80), -math.rad(80) }

local function shooting()
	local now = env.RivalsClock()
	local shotAt = env.RivalsShotAt
	if shotAt and now >= shotAt and now - shotAt < SHOT_HOLD then return true end
	local fire = env.RivalsFire
	local pressAt = fire and fire.lastPress
	return pressAt ~= nil and now >= pressAt and now - pressAt < SHOT_HOLD
end

local function currentLevel()
	local index = math.floor(tonumber(cfg.antiAimLevel) or 1)
	return LEVELS[math.max(1, math.min(#LEVELS, index))]
end

local lineParams = RaycastParams.new()
lineParams.FilterType = Enum.RaycastFilterType.Exclude
lineParams.IgnoreWater = true
lineParams.RespectCanCollide = true

local zoneParams = OverlapParams.new()
zoneParams.FilterType = Enum.RaycastFilterType.Exclude

local function hasTag(part, tag) return part:HasTag(tag) end

local function inKillZone(spot, character)
	zoneParams.FilterDescendantsInstances = { character }
	local ok, parts = pcall(function() return Workspace:GetPartBoundsInRadius(spot, KILL_RADIUS, zoneParams) end)
	if not ok then return true end
	for _, part in ipairs(parts or {}) do
		if part.Name == "Barriers" or hasTag(part, "OutOfBoundsPart") or hasTag(part, "KillBrick") then return true end
	end
	return false
end

local function sinkShift(character, base)
	lineParams.FilterDescendantsInstances = { character }
	local floor = Workspace:Raycast(base, Vector3.new(0, -JITTER_GROUND, 0), lineParams)
	if not floor or not floor.Position then return nil end
	local depth = SINK_MIN + math.random() * (SINK_MAX - SINK_MIN)
	local spot = Vector3.new(base.X, floor.Position.Y - depth, base.Z)
	if not (spot.Y >= Workspace.FallenPartsDestroyHeight + 50) then return nil end
	if inKillZone(spot, character) then return nil end
	return spot - base
end

local function spotOk(desync, bounds, root, character, base, shift)
	local spot = base + shift
	if bounds and not bounds:Allows(spot) then return false end
	lineParams.FilterDescendantsInstances = { character }
	if Workspace:Raycast(base + Vector3.new(0, 2, 0), shift, lineParams) then return false end
	if not Workspace:Raycast(spot, Vector3.new(0, -JITTER_GROUND, 0), lineParams) then return false end
	return desync:IsSafe(spot, root)
end

if env.RivalsAntiAimUnload then pcall(env.RivalsAntiAimUnload) end

local saved = nil
local angle = 0
local pausedUntil = 0
local recentEvents = {}
local lastWarn = -math.huge
local threatNow = nil

env.RivalsAntiAimStats = { threats = 0, meleeTime = 0, dodgeTime = 0, shieldTime = 0, sinkTime = 0 }
env.RivalsAntiAimMoved = false
local counters = env.RivalsAntiAimStats

local function levelPoint(position, point)
	local flat = Vector3.new(point.X, position.Y, point.Z)
	if (flat - position).Magnitude < 0.05 then return nil end
	return flat
end

local function ownShield()
	local targetState = env.RivalsTargetState
	if not targetState or not targetState.Read then return nil end
	local state = targetState:Read(player)
	if not state then return nil end
	if state.riotFront then return "front" end
	if state.riotBack then return "back" end
	return nil
end

local function dodgeShifts(base, threatPos, up)
	local away = Vector3.new(base.X - threatPos.X, 0, base.Z - threatPos.Z)
	local gap = away.Magnitude
	local ax, az = 1, 0
	if gap > 0.05 then ax, az = away.X / gap, away.Z / gap end
	local shifts = {}
	for _, turn in ipairs(DODGE_TURNS) do
		local c, s = math.cos(turn), math.sin(turn)
		local dx, dz = ax * c - az * s, ax * s + az * c
		local along = away.X * dx + away.Z * dz
		local length = math.max(0, -along + math.sqrt(math.max(0, along * along - gap * gap + MELEE_SAFE * MELEE_SAFE)))
		table.insert(shifts, Vector3.new(dx * length, up, dz * length))
	end
	local rise = math.sqrt(math.max(0, MELEE_SAFE * MELEE_SAFE - gap * gap)) - (base.Y - threatPos.Y)
	if rise < JITTER_GROUND - 4 then table.insert(shifts, Vector3.new(0, math.max(up, rise), 0)) end
	return shifts
end

local function onAnomaly(kind, amount)
	if env.RivalsEvent then env.RivalsEvent("안티에임 이상", kind .. " " .. string.format("%.1f", amount)) end
	local safety = env.RivalsSafety
	local now = safety and safety.clock() or env.RivalsClock()
	pausedUntil = now + PAUSE_AFTER
	if safety then
		if kind == "pull" then safety.jitterPulls = (safety.jitterPulls or 0) + 1
		else safety.jitterFlings = (safety.jitterFlings or 0) + 1 end
		safety.jitterScale = math.max(0.25, (safety.jitterScale or 1) * 0.6)
		safety.lastRecover = now
	end
	table.insert(recentEvents, now)
	while #recentEvents > 0 and now - recentEvents[1] > 30 do table.remove(recentEvents, 1) end
	if #recentEvents >= 5 and cfg.antiAimJitter then
		cfg.antiAimJitter = false
		warn("[YGJ] 안티에임 중 몸이 30초에 다섯 번 이상 밀렸다. 위치 튕김 오프")
	elseif now - lastWarn > 10 then
		lastWarn = now
		local what = kind == "pull" and string.format("몸이 %.1f스터드 끌려갔다", amount) or string.format("속도가 %.0f 튀었다", amount)
		warn("[YGJ] 안티에임 직후 " .. what .. ". 튕김 폭 줄이고 잠깐 쉰다")
	end
end

local function restore()
	if not saved then return end
	local current = saved
	saved = nil
	local root = current.root
	local anomaly, amount = nil, 0
	if root.Parent and current.displaced then
		local pulled = (root.CFrame.Position - current.displaced).Magnitude
		if pulled > PULL_GAP and pulled <= TELEPORT_GAP then anomaly, amount = "pull", pulled end
		if current.linear then
			local jump = (root.AssemblyLinearVelocity - current.linear).Magnitude
			if jump > FLING_SPEED then root.AssemblyLinearVelocity = current.linear anomaly, amount = "speed", jump end
		end
	end
	local teleported = root.Parent ~= nil and (root.CFrame.Position - current.cframe.Position).Magnitude > TELEPORT_GAP
	if root.Parent then
		if not teleported then root.CFrame = current.cframe end
		root.AssemblyAngularVelocity = current.angular
	end
	if anomaly then onAnomaly(anomaly, amount) end
end

local heartbeat = RunService.Heartbeat:Connect(function(dt)
	env.RivalsAntiAimMoved = false
	env.RivalsAntiAimFrameAt = env.RivalsClock()
	if not cfg.antiAim then return end
	local spin = cfg.antiAimSpin
	local jitter = cfg.antiAimJitter
	if not spin and not jitter then return end
	local desync = env.DesyncController
	if desync and desync.active then return end
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
	if not root or not humanoid or humanoid.Health <= 0 then return end
	restore()
	local step = dt or 1 / 60
	local level = currentLevel()
	local holding = shooting()
	saved = { root = root, cframe = root.CFrame, angular = root.AssemblyAngularVelocity, linear = root.AssemblyLinearVelocity }
	local position = saved.cframe.Position
	local controller = env.TargetController
	local safety = env.RivalsSafety

	local threat, threatDistance, threatRoot = nil, nil, nil
	if controller and controller.NearestMeleeThreat then
		threat, threatDistance, threatRoot = controller:NearestMeleeThreat(MELEE_NEAR)
	end
	if threat ~= threatNow then
		threatNow = threat
		if threat then
			counters.threats += 1
			if env.RivalsEvent then env.RivalsEvent("근접 위협", string.format("%s %.0f스터드", threat.Name, threatDistance)) end
		end
	end
	if threat and safety then safety.lastMelee = safety.clock() end
	local reacted = false

	if spin then
		local face = nil
		if threat then
			face = levelPoint(position, threatRoot.Position)
			reacted = face ~= nil
		else
			local shield = ownShield()
			local enemyRoot = nil
			if shield and controller and controller.NearestEnemy then enemyRoot = select(3, controller:NearestEnemy()) end
			if enemyRoot then
				local toward = shield == "front" and enemyRoot.Position or position + (position - enemyRoot.Position)
				face = levelPoint(position, toward)
				if face then counters.shieldTime += step end
			end
		end
		if face then root.CFrame = CFrame.lookAt(position, face)
		elseif level.tumble then
			root.CFrame = CFrame.new(position) * CFrame.Angles(math.random() * 2 * math.pi, math.random() * 2 * math.pi, math.random() * 2 * math.pi)
			root.AssemblyAngularVelocity = Vector3.new(0, level.spin, 0)
			if safety then safety.lastSpin = safety.clock() end
		else
			angle = (angle + level.spin * step) % (2 * math.pi)
			root.CFrame = CFrame.new(position) * CFrame.Angles(0, angle, 0)
			root.AssemblyAngularVelocity = Vector3.new(0, level.spin, 0)
			if safety then safety.lastSpin = safety.clock() end
		end
	end

	if jitter and not holding then
		local now = safety and safety.clock() or env.RivalsClock()
		if now >= pausedUntil and desync and desync.IsSafe then
			local bounds = env.RivalsBounds
			if threat then
				for _, shift in ipairs(dodgeShifts(position, threatRoot.Position, level.upMin)) do
					if spotOk(desync, bounds, root, character, position, shift) then
						root.CFrame = root.CFrame + shift
						env.RivalsAntiAimMoved = true
						counters.dodgeTime += step
						reacted = true
						if safety then safety.lastJitter = safety.clock() end
						break
					end
				end
			else
				local distance = controller and controller.NearestEnemyDistance and controller:NearestEnemyDistance()
				local sunk = level.sink and distance ~= nil and distance <= SINK_RANGE and sinkShift(character, position)
				if sunk then
					root.CFrame = root.CFrame + sunk
					env.RivalsAntiAimMoved = true
					counters.sinkTime += step
					if safety then safety.lastJitter = safety.clock() end
				elseif distance and distance <= JITTER_RANGE then
					local scale = safety and safety.jitterScale or 1
					local up = level.upMin + math.random() * (level.upMax - level.upMin) * scale
					local candidates = {}
					if bounds and scale >= 0.5 and level.farChance > 0 and math.random() < level.farChance then
						local an = math.random() * 2 * math.pi
						local radius = (level.farMin + math.random() * (level.farMax - level.farMin)) * scale
						table.insert(candidates, Vector3.new(math.cos(an) * radius, up, math.sin(an) * radius))
					end
					table.insert(candidates, Vector3.new((math.random() * 2 - 1) * level.side * scale, up, (math.random() * 2 - 1) * level.side * scale))
					table.insert(candidates, Vector3.new(0, 2, 0))
					local shift = nil
					for _, candidate in ipairs(candidates) do
						if spotOk(desync, bounds, root, character, position, candidate) then shift = candidate break end
					end
					if shift then
						root.CFrame = root.CFrame + shift
						env.RivalsAntiAimMoved = true
						if safety then safety.lastJitter = safety.clock() end
					end
				end
			end
		end
	end
	if reacted then counters.meleeTime += step end
	if saved then saved.displaced = root.CFrame.Position end
end)

pcall(function() RunService:UnbindFromRenderStep(RESTORE_NAME) end)
if not pcall(function() RunService:BindToRenderStep(RESTORE_NAME, Enum.RenderPriority.First.Value, restore) end) then
	heartbeat:Disconnect()
	warn("[YGJ] antiaim: 복원 연결 실패. 안티에임 오프")
	return
end

	heartbeat:Disconnect()
	env.RivalsAntiAimMoved = false
	pcall(function() RunService:UnbindFromRenderStep(RESTORE_NAME) end)
	restore()

local function finite(v) return type(v) == "number" and v == v and v ~= math.huge and v ~= -math.huge end

local function sanitize(raw)
	local out = { features = {}, window = {} }
	if type(raw) ~= "table" then return out end
	local f = raw.features
	if type(f) == "table" then
		for _, key in ipairs(FEATURE_KEYS) do
			if type(f[key]) == "boolean" then out.features[key] = f[key] end
		end
		if finite(f.antiAimLevel) and f.antiAimLevel >= 1 and f.antiAimLevel <= 4 then
			out.features.antiAimLevel = math.floor(f.antiAimLevel + 0.5)
		end
	end
	local w = raw.window
	if type(w) == "table" and w.layout == LAYOUT then
		if finite(w.x) and finite(w.y) then out.window.x, out.window.y = w.x, w.y end
		if finite(w.scale) then out.window.scale = math.clamp(w.scale, MIN_SCALE, MAX_SCALE) end
	end
	return out
end

local function statusContent()
	local stats = env.RivalsStats
	if not stats then return "아직 기록 없음" end
	local lines = {}
	local judged = (stats.windowHits or 0) + (stats.windowMisses or 0)
	if judged > 0 then
		table.insert(lines, string.format("명중 %d/%d (%d%%)", stats.windowHits, judged, math.floor(100 * stats.windowHits / judged + 0.5)))
	else
		table.insert(lines, string.format("명중 %d/%d", stats.hits or 0, (stats.hits or 0) + (stats.misses or 0)))
	end
	if (stats.headTotal or 0) > 0 then
		table.insert(lines, string.format("헤드 %d/%d (%d%%)", stats.headHits, stats.headTotal, math.floor(100 * stats.headHits / stats.headTotal + 0.5)))
	end
	local check = env.RivalsCheck
	if check then
		local bad = 0
		for _, item in ipairs(check.items) do if not item.ok then bad += 1 end end
		table.insert(lines, bad == 0 and "에임 점검 정상" or ("에임 점검 문제 " .. bad))
	end
	local safety = env.RivalsSafety
	if safety then
		if safety.rangeScale < 1 then table.insert(lines, string.format("사거리 x%.2f", safety.rangeScale)) end
		if safety.jitterScale < 1 then table.insert(lines, string.format("튕김 x%.2f", safety.jitterScale)) end
	end
	return table.concat(lines, "\n")
end

local function recordContent()
	local lines = { env.RivalsVersion or "체크포인트 ?" }
	local stats = env.RivalsStats
	if stats and type(stats.log) == "table" then

	local events = env.RivalsEvents
	if events and type(events.list) == "table" then table.insert(lines, "이벤트 " .. #events.list) end
	return table.concat(lines, "\n")
end

	new("Frame", { Position = UDim2.fromOffset(0, 14 * S), Size = px(3, 24), BackgroundColor3 = row.color or C.neon, BorderSizePixel = 0, ZIndex = 5 }, frame)
	corner(new("Frame", { Position = UDim2.fromOffset(0, 14 * S), Size = px(3, 24), BackgroundColor3 = row.color or C.neon, BorderSizePixel = 0, ZIndex = 5 }, frame), 2)
	text(frame, { Position = at(18, 0), Size = px(120, 52), TextSize = 14, Text = row.label, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 5 })
	local width, count = 240, #row.options
	local seg = new("Frame", { AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14 * S, 0.5, 0), Size = px(width, 28), BackgroundColor3 = C.cardHi, BorderSizePixel = 0, ZIndex = 5 }, frame)
	corner(seg, 8)
	local slot = (width - 4) / count
	local chip = new("Frame", { Position = UDim2.fromOffset(2 * S, 2 * S), Size = UDim2.fromOffset(slot * S, 24 * S), BackgroundColor3 = row.color or C.neon, BorderSizePixel = 0, ZIndex = 5 }, seg)
	corner(chip, 6)
	local buttons, shown = {}, nil
	local function refresh()
		local want = row.get()
		if shown == want then return end
		shown = want
		for index, entry in ipairs(buttons) do
			local on = entry.value == want
			entry.btn.Font = on and FONT_BOLD or FONT
			entry.btn.TextColor3 = on and C.bg0 or C.dim
			if on then tween(chip, 0.15, { Position = UDim2.fromOffset((2 + (index - 1) * slot) * S, 2 * S) }) end
		end
	end
	for index, option in ipairs(row.options) do
		local b = new("TextButton", {
			Text = option[2], AutoButtonColor = false, BorderSizePixel = 0, BackgroundTransparency = 1,
			Position = UDim2.fromOffset((2 + (index - 1) * slot) * S, 2 * S), Size = UDim2.fromOffset(slot * S, 24 * S),
			Font = FONT, TextSize = 12, TextColor3 = C.dim, ZIndex = 6,
		}, seg)
		b.MouseButton1Click:Connect(function() row.set(option[1]) shown = nil refresh() end)
		table.insert(buttons, { btn = b, value = option[1] })
	end
	table.insert(refreshers, refresh)
	refresh()
end

local function addButton(parent, order, row)
	local btn = new("TextButton", { Text = "", AutoButtonColor = false, BackgroundColor3 = C.card, BorderSizePixel = 0, LayoutOrder = order, Size = px(BODY_W, 46), ZIndex = 4 }, parent)
	corner(btn, 10)
	stroke(btn, C.line, 1, 0)
	text(btn, { Position = at(18, 0), Size = px(BODY_W - 36, 46), TextSize = 14, Text = row.label, TextColor3 = row.color or C.neon, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 5 })
	btn.MouseEnter:Connect(function() tween(btn, 0.12, { BackgroundColor3 = C.cardHi }) end)
	btn.MouseLeave:Connect(function() tween(btn, 0.12, { BackgroundColor3 = C.card }) end)
	btn.MouseButton1Click:Connect(function()
		local ok, err = pcall(row.run)
		if not ok then warn("[YGJ] " .. tostring(err)) end
		if refreshAll then refreshAll() end
	end)
end

local function addText(parent, order, row, card)
	local frame = new("Frame", {
		BackgroundColor3 = card and C.card or C.cardHi, BackgroundTransparency = card and 0 or 0.3, BorderSizePixel = 0,
		LayoutOrder = order, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, ZIndex = 4,
	}, parent)
	corner(frame, 10)
	if card then stroke(frame, C.line, 1, 0) end
	new("UIPadding", { PaddingTop = UDim.new(0, 12 * S), PaddingBottom = UDim.new(0, 12 * S), PaddingLeft = UDim.new(0, 16 * S), PaddingRight = UDim.new(0, 16 * S) }, frame)
	local label = text(frame, {
		Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, TextSize = card and 13 or 11,
		TextTransparency = card and 0 or 0.35, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top,
		TextWrapped = true, LineHeight = 1.25, ZIndex = 5,
	})
	local shown = nil
	local function refresh()
		local value = row.text()
		if value ~= shown then shown = value label.Text = value end
	end
	table.insert(refreshers, refresh)
	refresh()
end

local bodyPages = {}
for _, tab in ipairs(TABS) do
	local page = new("ScrollingFrame", {
		Name = "Page_" .. tab.id, BackgroundTransparency = 1, BorderSizePixel = 0, Visible = false,
		Position = at(0, 0), Size = UDim2.fromScale(1, 1), CanvasSize = UDim2.fromOffset(0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollBarThickness = 3, ScrollBarImageColor3 = C.neon, ScrollBarImageTransparency = 0.5,
		ZIndex = 4,
	}, body)
	new("UIListLayout", { Padding = UDim.new(0, 9 * S), SortOrder = Enum.SortOrder.LayoutOrder }, page)
	bodyPages[tab.id] = page
	for order, row in ipairs(tab.rows) do
		if row.type == "toggle" then addToggle(page, order, row)
		elseif row.type == "segment" then addSegment(page, order, row)
		elseif row.type == "button" then addButton(page, order, row)
		else addText(page, order, row, row.type == "info") end
	end
end

local currentTab = nil
local function selectTab(id)
	currentTab = id
	for key, data in pairs(tabButtons) do
		local on = key == id
		tween(data.btn, 0.15, { BackgroundTransparency = on and 0 or 1 })
		data.label.TextColor3 = on and data.color or C.dim
		data.label.Font = on and FONT_BOLD or FONT_MED
		data.ind.Visible = on
	end
	for key, page in pairs(bodyPages) do page.Visible = key == id end
	if refreshAll then refreshAll() end
end

for id, data in pairs(tabButtons) do
	data.btn.MouseButton1Click:Connect(function() selectTab(id) end)
	data.btn.MouseEnter:Connect(function()
		if currentTab ~= id then tween(data.btn, 0.12, { BackgroundColor3 = C.cardHi, BackgroundTransparency = 0 }) end
	end)
	data.btn.MouseLeave:Connect(function()
		if currentTab ~= id then tween(data.btn, 0.12, { BackgroundTransparency = 1 }) end
	end)
end

function refreshAll()
	for _, refresh in ipairs(refreshers) do refresh() end
end

local isOpen = false
local function setOpen(open)
	isOpen = open
	pcall(function() ui.gui.Enabled = open end)
	cfg.menuOpen = open
	setCursor(open)
	if open then
		selectTab(currentTab or "aim")
		refreshAll()
	end
	if dirty then save() end
end

local drag = nil
local function pressed(input)
	local k = input.UserInputType
	return k == Enum.UserInputType.MouseButton1 or k == Enum.UserInputType.Touch
end
local function moved(input)
	local k = input.UserInputType
	return k == Enum.UserInputType.MouseMovement or k == Enum.UserInputType.Touch
end

titlebar.InputBegan:Connect(function(input)
	if pressed(input) and not drag then
		local p = input.Position
		drag = { px = p.X, py = p.Y, x0 = win.x, y0 = win.y }
	end
end)

table.insert(connections, UserInputService.InputChanged:Connect(function(input)
	if not drag or not moved(input) then return end
	local p = input.Position
	win.x = drag.x0 + (p.X - drag.px)
	win.y = drag.y0 + (p.Y - drag.py)
	applyWindow()
end))

table.insert(connections, UserInputService.InputEnded:Connect(function(input)
	if not pressed(input) or not drag then return end
	drag = nil
	rememberWindow()
end))

placeWindow(true)
minimized = false
setCursor(false)
selectTab("aim")
ui.gui.Enabled = false

local pollClock = 0
local lastW, lastH = viewport()
table.insert(connections, RunService.Heartbeat:Connect(function(dt)
	pollClock += dt or 0
	if dirty and env.RivalsClock() - lastChange >= SAVE_AFTER then save() end
	if pollClock >= 0.15 then
		pollClock = 0
		if isOpen then refreshAll() end
		local w, h = viewport()
		if w ~= lastW or h ~= lastH then
			lastW, lastH = w, h
			applyWindow()
		end
	end
end))

table.insert(connections, UserInputService.InputBegan:Connect(function(input, processed)
	if processed then return end
	if input.KeyCode == TOGGLE_KEY then setOpen(not isOpen)
	elseif input.KeyCode == UNLOAD_KEY then unloadAll() end
end))

env.RivalsMenu = {
	gui = ui.gui,
	window = win,
	settings = saved,
	Open = function(_, open) setOpen(open) end,
	Toggle = function() setOpen(not isOpen) end,
	Save = function() return save() end,
	Touch = function() touch() end,
	ResetWindow = function() resetWindow() end,
	Features = FEATURE_KEYS,
}

env.RivalsMenuUnload = function()
	for _, connection in ipairs(connections) do connection:Disconnect() end
	if dirty then save() end
	cfg.menuOpen = false
	setCursor(false)
	env.RivalsMenu = nil
	ui.gui:Destroy()
end

env.RivalsUnloadAll = function()
	for _, name in ipairs({ "RivalsAutoFireUnload", "RivalsAntiAimUnload", "RivalsEspUnload", "RivalsAimUnload", "RivalsBaseUnload" }) do
		local unload = env[name]
		if type(unload) == "function" then pcall(unload) end
	end
	warn("[YGJ] 전부 껐다. 다시 쓰려면 스크립트를 재실행해라")
	pcall(env.RivalsMenuUnload)
end
